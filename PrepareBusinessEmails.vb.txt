' Paste this entire file into ONE Invoke Code activity.
' DisplayName: MAIL - Prepare Business Emails
' Language: VBNet | ContinueOnError: False
'
' Invoke Code > Edit Arguments:
' in_Config       | In  | System.Collections.Generic.Dictionary(Of String, Object)
' in_ReportPeriod | In  | System.String (Monthly: yyyy-MM or yyyy-MM-dd; Daily: yyyy-MM-dd)
' in_Frequency    | In  | System.String (Monthly or Daily)
' out_MailPlan    | Out | System.Data.DataTable (map to dtMailPlan)
'
' Required Config keys:
' OutputRootFolder, CrmGroupFilePath, FileNamePrefix, EmailSubject, EmailBodyHtml,
' Email_SuccessMessage, Email_PrimaryColor, Email_AccentColor,
' Email_DateFormat, Email_TimeFormat.
' Optional Config keys: CrmGroupSheetName (blank = first worksheet),
' OutlookAccount (used by Send Outlook Mail Message outside this code).
'
' Put the complete success HTML template in EmailBodyHtml.
' Email_SuccessMessage is trusted HTML; inserted group/file/period values are encoded.
' {{REPORT_NAME}} is the current attachment filename without its extension.
' {{RUN_DATE}} / {{RUN_TIME}} use the robot computer's local run timestamp.
' Monthly folder: <root>/<yyyy>/<MMM>; Daily folder: <root>/<yyyy>/<MMM>/<dd>.
' Native Outlook submission, state reservation, logs and Try Catch remain in the XAML.
' This code prepares and validates the complete plan BEFORE any email is submitted.

out_MailPlan = Nothing
Dim inv As System.Globalization.CultureInfo = System.Globalization.CultureInfo.InvariantCulture
Dim cmp As System.StringComparer = System.StringComparer.OrdinalIgnoreCase
Dim comparison As System.StringComparison = System.StringComparison.OrdinalIgnoreCase

If in_Config Is Nothing Then Throw New System.IO.InvalidDataException("Config is Nothing.")
Dim configText As System.Func(Of String, String) =
    Function(key As String) As String
        Dim configValue As Object = Nothing
        If Not in_Config.TryGetValue(key, configValue) OrElse
           configValue Is Nothing OrElse configValue Is System.DBNull.Value Then
            Return String.Empty
        End If
        Return System.Convert.ToString(configValue, inv)
    End Function

For Each key As String In New String() {
    "OutputRootFolder", "CrmGroupFilePath", "FileNamePrefix", "EmailSubject", "EmailBodyHtml",
    "Email_SuccessMessage", "Email_PrimaryColor", "Email_AccentColor",
    "Email_DateFormat", "Email_TimeFormat"
}
    If String.IsNullOrWhiteSpace(configText(key)) Then
        Throw New System.IO.InvalidDataException("Missing Config value: " & key)
    End If
Next

Dim frequency As String = If(in_Frequency, String.Empty).Trim()
If Not String.Equals(frequency, "Monthly", comparison) AndAlso Not String.Equals(frequency, "Daily", comparison) Then
    Throw New System.IO.InvalidDataException("in_Frequency must be Monthly or Daily.")
End If
Dim isDaily As Boolean = String.Equals(frequency, "Daily", comparison)
Dim periodText As String = If(in_ReportPeriod, String.Empty).Trim()
Dim periodDate As System.DateTime
Dim periodFormats As String() = If(isDaily, New String() {"yyyy-MM-dd"}, New String() {"yyyy-MM", "yyyy-MM-dd"})
If Not System.DateTime.TryParseExact(periodText, periodFormats, inv,
    System.Globalization.DateTimeStyles.None, periodDate) Then
    Throw New System.IO.InvalidDataException(If(isDaily, "Daily in_ReportPeriod must be yyyy-MM-dd.", "Monthly in_ReportPeriod must be yyyy-MM or yyyy-MM-dd."))
End If
Dim periodLabel As String = periodDate.ToString(If(isDaily, "dd MMMM yyyy", "MMMM yyyy"), inv)
Dim prefix As String = configText("FileNamePrefix").Trim()
If prefix.IndexOfAny(("<>:""/\|?*").ToCharArray()) >= 0 Then
    Throw New System.IO.InvalidDataException("FileNamePrefix contains an invalid filename character.")
End If
For Each ch As Char In prefix
    If Char.IsControl(ch) Then Throw New System.IO.InvalidDataException("FileNamePrefix contains a control character.")
Next
For Each key As String In New String() {"OutputRootFolder", "CrmGroupFilePath"}
    Dim pathValue As String = configText(key).Trim()
    If Not System.IO.Path.IsPathRooted(pathValue) OrElse
       (System.Environment.OSVersion.Platform = System.PlatformID.Win32NT AndAlso
        System.IO.Path.GetPathRoot(pathValue).Length < 3) Then
        Throw New System.IO.InvalidDataException(key & " must be an absolute local or UNC path.")
    End If
Next
Dim crmPath As String = System.IO.Path.GetFullPath(configText("CrmGroupFilePath").Trim())
Dim periodFolder As String = System.IO.Path.Combine(System.IO.Path.GetFullPath(configText("OutputRootFolder").Trim()),
    periodDate.ToString("yyyy", inv), periodDate.ToString("MMM", inv))
If isDaily Then periodFolder = System.IO.Path.Combine(periodFolder, periodDate.ToString("dd", inv))
If Not System.IO.File.Exists(crmPath) Then Throw New System.IO.InvalidDataException("CRM Group workbook does not exist: " & crmPath)
If Not System.IO.Directory.Exists(periodFolder) Then Throw New System.IO.InvalidDataException("Reporting period folder does not exist: " & periodFolder)

Dim headerMap As System.Func(Of System.Array, System.Collections.Generic.Dictionary(Of String, Integer)) =
    Function(values As System.Array) As System.Collections.Generic.Dictionary(Of String, Integer)
        If values Is Nothing OrElse values.Rank <> 2 Then
            Throw New System.IO.InvalidDataException("CRM must have headers in row 1 and data below them.")
        End If
        Dim headers As New System.Collections.Generic.Dictionary(Of String, Integer)(cmp)
        For c As Integer = values.GetLowerBound(1) To values.GetUpperBound(1)
            Dim label As String = System.Text.RegularExpressions.Regex.Replace(
                System.Convert.ToString(values.GetValue(values.GetLowerBound(0), c), inv).Trim(), "\s+", " ")
            If Not String.Equals(label, "Group Name", comparison) AndAlso
               Not String.Equals(label, "Business Emails", comparison) Then Continue For
            If headers.ContainsKey(label) Then Throw New System.IO.InvalidDataException("Duplicate CRM header: " & label)
            headers.Add(label, c - values.GetLowerBound(1) + 1)
        Next
        For Each label As String In New String() {"Group Name", "Business Emails"}
            If Not headers.ContainsKey(label) Then Throw New System.IO.InvalidDataException("Missing CRM header in row 1: " & label)
        Next
        Return headers
    End Function

Dim readCrm As System.Func(Of System.Array) =
    Function() As System.Array
        Dim held As New System.Collections.Generic.List(Of Object)()
        Dim app As Object = Nothing
        Dim book As Object = Nothing
        Dim readFailed As Boolean = False
        Dim cleanupFailure As System.Exception = Nothing
        Dim xl As System.Func(Of Object, String, Microsoft.VisualBasic.CallType, Object(), Object) =
            Function(target As Object, member As String, kind As Microsoft.VisualBasic.CallType, args As Object()) As Object
                Dim value As Object = Microsoft.VisualBasic.Interaction.CallByName(target, member, kind, args)
                If value IsNot Nothing AndAlso System.Runtime.InteropServices.Marshal.IsComObject(value) Then held.Add(value)
                Return value
            End Function
        Dim getKind As Microsoft.VisualBasic.CallType = Microsoft.VisualBasic.CallType.Get
        Dim letKind As Microsoft.VisualBasic.CallType = Microsoft.VisualBasic.CallType.Let
        Dim methodKind As Microsoft.VisualBasic.CallType = Microsoft.VisualBasic.CallType.Method
        Try
            app = Microsoft.VisualBasic.Interaction.CreateObject("Excel.Application")
            held.Add(app)
            xl(app, "Visible", letKind, New Object() {False})
            xl(app, "DisplayAlerts", letKind, New Object() {False})
            xl(app, "EnableEvents", letKind, New Object() {False})
            xl(app, "AskToUpdateLinks", letKind, New Object() {False})
            xl(app, "AutomationSecurity", letKind, New Object() {3})
            Dim books As Object = xl(app, "Workbooks", getKind, New Object() {})
            book = xl(books, "Open", methodKind, New Object() {crmPath, 0, True, System.Type.Missing,
                "", "", True, System.Type.Missing, System.Type.Missing, False, False, System.Type.Missing, False, False, 0})
            Dim sheets As Object = xl(book, "Worksheets", getKind, New Object() {})
            Dim sheetName As String = configText("CrmGroupSheetName").Trim()
            Dim sheet As Object = Nothing
            If sheetName.Length = 0 Then
                sheet = xl(sheets, "Item", getKind, New Object() {1})
            Else
                Dim sheetCount As Integer = System.Convert.ToInt32(xl(sheets, "Count", getKind, New Object() {}), inv)
                For i As Integer = 1 To sheetCount
                    Dim candidate As Object = xl(sheets, "Item", getKind, New Object() {i})
                    If String.Equals(System.Convert.ToString(xl(candidate, "Name", getKind, New Object() {}), inv), sheetName, comparison) Then
                        sheet = candidate
                        Exit For
                    End If
                Next
                If sheet Is Nothing Then Throw New System.IO.InvalidDataException("CRM worksheet does not exist: " & sheetName)
            End If
            Dim firstRow As Object = xl(sheet, "Range", getKind, New Object() {"1:1"})
            Dim headers As System.Collections.Generic.Dictionary(Of String, Integer) = headerMap(
                TryCast(xl(firstRow, "Value2", getKind, New Object() {}), System.Array))
            Dim cells As Object = xl(sheet, "Cells", getKind, New Object() {})
            Dim rows As Object = xl(sheet, "Rows", getKind, New Object() {})
            Dim rowLimit As Integer = System.Convert.ToInt32(xl(rows, "Count", getKind, New Object() {}), inv)
            Dim lastRow As Integer = 1
            For Each col As Integer In New Integer() {headers("Group Name"), headers("Business Emails")}
                Dim bottom As Object = xl(cells, "Item", getKind, New Object() {rowLimit, col})
                Dim lastCell As Object = xl(bottom, "End", getKind, New Object() {-4162})
                lastRow = System.Math.Max(lastRow, System.Convert.ToInt32(xl(lastCell, "Row", getKind, New Object() {}), inv))
            Next
            If lastRow < 2 Then Throw New System.IO.InvalidDataException("CRM contains headers but no routing rows.")
            Dim lastCol As Integer = System.Math.Max(headers("Group Name"), headers("Business Emails"))
            Dim firstCell As Object = xl(cells, "Item", getKind, New Object() {1, 1})
            Dim finalCell As Object = xl(cells, "Item", getKind, New Object() {lastRow, lastCol})
            Dim dataRange As Object = xl(sheet, "Range", getKind, New Object() {firstCell, finalCell})
            Return TryCast(xl(dataRange, "Value2", getKind, New Object() {}), System.Array)
        Catch
            readFailed = True
            Throw
        Finally
            If book IsNot Nothing Then
                Try
                    Microsoft.VisualBasic.Interaction.CallByName(book, "Close", methodKind, False)
                Catch ex As System.Exception
                    cleanupFailure = ex
                End Try
            End If
            If app IsNot Nothing Then
                Try
                    Microsoft.VisualBasic.Interaction.CallByName(app, "Quit", methodKind)
                Catch ex As System.Exception
                    If cleanupFailure Is Nothing Then cleanupFailure = ex
                End Try
            End If
            For i As Integer = held.Count - 1 To 0 Step -1
                Try
                    If System.Runtime.InteropServices.Marshal.IsComObject(held(i)) Then
                        System.Runtime.InteropServices.Marshal.ReleaseComObject(held(i))
                    End If
                Catch ex As System.Exception
                    If cleanupFailure Is Nothing Then cleanupFailure = ex
                End Try
            Next
            If Not readFailed AndAlso cleanupFailure IsNot Nothing Then
                Throw New System.IO.IOException("The CRM Excel instance could not close cleanly.", cleanupFailure)
            End If
        End Try
    End Function

Dim raw As System.Array = readCrm()
Dim columns As System.Collections.Generic.Dictionary(Of String, Integer) = headerMap(raw)
Dim groupColumn As Integer = raw.GetLowerBound(1) + columns("Group Name") - 1
Dim emailColumn As Integer = raw.GetLowerBound(1) + columns("Business Emails") - 1
Dim groupRows As New System.Collections.Generic.Dictionary(Of String, System.Collections.Generic.List(Of Integer))(cmp)
For r As Integer = raw.GetLowerBound(0) + 1 To raw.GetUpperBound(0)
    Dim groupValue As Object = raw.GetValue(r, groupColumn)
    Dim group As String = System.Convert.ToString(groupValue, inv).Trim()
    Dim emails As String = System.Convert.ToString(raw.GetValue(r, emailColumn), inv).Trim()
    Dim excelRow As Integer = r - raw.GetLowerBound(0) + 1
    If group.Length = 0 Then
        If emails.Length > 0 Then Throw New System.IO.InvalidDataException("CRM row " & excelRow.ToString(inv) & ": Business Emails has a value but Group Name is empty.")
        Continue For
    End If
    If Not TypeOf groupValue Is String Then Throw New System.IO.InvalidDataException("CRM row " & excelRow.ToString(inv) & ": Group Name must be text.")
    If Not groupRows.ContainsKey(group) Then groupRows.Add(group, New System.Collections.Generic.List(Of Integer)())
    groupRows(group).Add(r)
Next
If groupRows.Count = 0 Then Throw New System.IO.InvalidDataException("CRM contains no group names.")

Dim subjectTemplate As String = configText("EmailSubject").Trim()
If subjectTemplate.IndexOfAny(New Char() {ChrW(13), ChrW(10)}) >= 0 Then
    Throw New System.IO.InvalidDataException("EmailSubject must be one line.")
End If
Dim bodyTemplate As String = configText("EmailBodyHtml")
Dim RunTimestamp As System.DateTime = System.DateTime.Now
Dim html As System.Func(Of String, String) = Function(value As String) System.Net.WebUtility.HtmlEncode(value)
Dim plan As New System.Data.DataTable("MailPlan")
For Each name As String In New String() {"GroupName", "To", "AttachmentPath", "Subject", "BodyHtml", "StatePath", "Status"}
    plan.Columns.Add(name, GetType(String))
Next
Dim stateFolder As String = System.IO.Path.Combine(periodFolder, "_logs", "EmailDispatch")
Dim candidates As New System.Collections.Generic.List(Of String)()
For Each filePath As String In System.IO.Directory.GetFiles(periodFolder, "*", System.IO.SearchOption.TopDirectoryOnly)
    Dim fileName As String = System.IO.Path.GetFileName(filePath)
    If fileName.StartsWith("~$", System.StringComparison.Ordinal) Then Continue For
    If String.Equals(System.IO.Path.GetExtension(filePath), ".xlsx", comparison) AndAlso
       fileName.StartsWith(prefix & " - ", comparison) Then candidates.Add(filePath)
Next
candidates.Sort(cmp)
If candidates.Count = 0 Then Throw New System.IO.InvalidDataException("No matching group .xlsx files exist in: " & periodFolder)
Dim matched As New System.Collections.Generic.HashSet(Of String)(cmp)
Dim alreadySubmitted As Integer = 0
For Each filePath As String In candidates
    Dim stem As String = System.IO.Path.GetFileNameWithoutExtension(filePath)
    Dim group As String = stem.Substring(prefix.Length + 3)
    If group.Length = 0 OrElse Not groupRows.ContainsKey(group) Then
        Throw New System.IO.InvalidDataException("Output file has no exact CRM Group Name match: " & System.IO.Path.GetFileName(filePath))
    End If
    If Not matched.Add(group) Then Throw New System.IO.InvalidDataException("More than one output workbook matches group: " & group)
    For Each ch As Char In group
        If Char.IsControl(ch) Then Throw New System.IO.InvalidDataException("A group filename contains a control character.")
    Next
    Dim recipients As New System.Collections.Generic.HashSet(Of String)(cmp)
    For Each r As Integer In groupRows(group)
        Dim cell As Object = raw.GetValue(r, emailColumn)
        Dim value As String = System.Convert.ToString(cell, inv).Trim()
        If value.Length = 0 Then Continue For
        Dim excelRow As Integer = r - raw.GetLowerBound(0) + 1
        If Not TypeOf cell Is String Then Throw New System.IO.InvalidDataException("CRM row " & excelRow.ToString(inv) & ": Business Emails must contain text addresses.")
        For Each part As String In System.Text.RegularExpressions.Regex.Split(value, "[;,\r\n]+")
            Dim email As String = part.Trim()
            If email.Length = 0 Then Continue For
            Try
                Dim address As New System.Net.Mail.MailAddress(email)
                If address.DisplayName.Length > 0 OrElse Not String.Equals(address.Address, email, comparison) Then Throw New System.FormatException()
                recipients.Add(address.Address)
            Catch ex As System.FormatException
                Throw New System.IO.InvalidDataException("CRM row " & excelRow.ToString(inv) & ": invalid Business Emails; use email addresses separated by semicolons.")
            End Try
        Next
    Next
    If recipients.Count = 0 Then Throw New System.IO.InvalidDataException("Business Emails is empty for output group: " & group)
    Dim statePath As String = System.IO.Path.Combine(stateFolder, "mail-" & group & ".status")
    If System.IO.File.Exists(statePath) Then
        If String.Equals(System.IO.File.ReadAllText(statePath).Trim(), "Submitted", System.StringComparison.Ordinal) Then
            alreadySubmitted += 1
            Continue For
        End If
        Throw New System.IO.InvalidDataException("A previous email attempt needs review before rerun: " & System.IO.Path.GetFileName(statePath))
    End If
    Using stream As New System.IO.FileStream(filePath, System.IO.FileMode.Open, System.IO.FileAccess.Read, System.IO.FileShare.Read)
        If stream.Length = 0 Then Throw New System.IO.InvalidDataException("Output workbook is empty: " & System.IO.Path.GetFileName(filePath))
    End Using
    Dim addresses As New System.Collections.Generic.List(Of String)(recipients)
    addresses.Sort(cmp)
    Dim subject As String = subjectTemplate.Replace("{{GROUP_NAME}}", group).Replace("{{REPORT_PERIOD}}", periodLabel)
    Dim body As String = bodyTemplate.
        Replace("{{STATUS_TEXT}}", "COMPLETED").
        Replace("{{RESULT_TEXT}}", "Download completed successfully").
        Replace("{{MESSAGE_HTML}}", configText("Email_SuccessMessage")).
        Replace("{{PRIMARY_COLOR}}", configText("Email_PrimaryColor")).
        Replace("{{ACCENT_COLOR}}", configText("Email_AccentColor")).
        Replace("{{RUN_DATE}}", RunTimestamp.ToString(configText("Email_DateFormat"), inv)).
        Replace("{{RUN_TIME}}", RunTimestamp.ToString(configText("Email_TimeFormat"), inv)).
        Replace("{{FOLDER_1}}", html(filePath)).
        Replace("{{REPORT_NAME}}", html(System.IO.Path.GetFileNameWithoutExtension(filePath))).
        Replace("{{GROUP_NAME}}", html(group)).
        Replace("{{REPORT_PERIOD}}", html(periodLabel))
    If subject.Contains("{{") OrElse subject.Contains("}}") OrElse body.Contains("{{") OrElse body.Contains("}}") Then
        Throw New System.IO.InvalidDataException("EmailSubject / EmailBodyHtml contains an unsupported or misspelled placeholder.")
    End If
    plan.Rows.Add(New Object() {group, String.Join(";", addresses.ToArray()), filePath, subject, body, statePath, "Ready"})
Next
System.IO.Directory.CreateDirectory(stateFolder)
plan.ExtendedProperties("AlreadySubmitted") = alreadySubmitted
plan.ExtendedProperties("PeriodFolder") = periodFolder
out_MailPlan = plan
