' Late-bound COM through CallByName also works with Option Strict On.
' Retry only COM calls Excel explicitly rejected as busy; never retry a whole export here.
Dim xl As System.Func(Of Object, String, Microsoft.VisualBasic.CallType, Object(), Object) =
    Function(target As Object, member As String, callType As Microsoft.VisualBasic.CallType, args As Object()) As Object
        For attempt As Integer = 1 To 3
            Try
                Return Microsoft.VisualBasic.Interaction.CallByName(target, member, callType, args)
            Catch ex As System.Exception
                Dim cause As System.Exception = ex
                While TypeOf cause Is System.Reflection.TargetInvocationException AndAlso cause.InnerException IsNot Nothing
                    cause = cause.InnerException
                End While
                Dim comError As System.Runtime.InteropServices.COMException =
                    TryCast(cause, System.Runtime.InteropServices.COMException)
                If comError Is Nothing OrElse
                   (comError.ErrorCode <> -2147418111 AndAlso comError.ErrorCode <> -2147417846) OrElse
                   attempt = 3 Then
                    Throw
                End If
                System.Threading.Thread.Sleep(250 * attempt)
            End Try
        Next
        Throw New System.InvalidOperationException("Excel COM retry limit reached.")
    End Function

' Invoke worksheet Copy/Move directly through COM with explicit argument names.
' No Missing/After positional arguments and no dependency on the active destination.
Dim xlMethod As System.Func(Of Object, String, Object(), String(), Object) =
    Function(target As Object, member As String, args As Object(), argumentNames As String()) As Object
        For attempt As Integer = 1 To 3
            Try
                Return target.GetType().InvokeMember(member,
                    System.Reflection.BindingFlags.InvokeMethod Or System.Reflection.BindingFlags.Public Or
                    System.Reflection.BindingFlags.Instance Or System.Reflection.BindingFlags.OptionalParamBinding,
                    Nothing, target, args, Nothing,
                    System.Globalization.CultureInfo.GetCultureInfo("en-US"), argumentNames)
            Catch ex As System.Exception
                Dim cause As System.Exception = ex
                While TypeOf cause Is System.Reflection.TargetInvocationException AndAlso cause.InnerException IsNot Nothing
                    cause = cause.InnerException
                End While
                Dim comError As System.Runtime.InteropServices.COMException =
                    TryCast(cause, System.Runtime.InteropServices.COMException)
                If comError Is Nothing OrElse
                   (comError.ErrorCode <> -2147418111 AndAlso comError.ErrorCode <> -2147417846) OrElse
                   attempt = 3 Then
                    Throw
                End If
                System.Threading.Thread.Sleep(250 * attempt)
            End Try
        Next
        Throw New System.InvalidOperationException("Excel COM method retry limit reached.")
    End Function

Dim releaseCom As System.Action(Of Object) =
    Sub(obj As Object)
        If obj IsNot Nothing AndAlso System.Runtime.InteropServices.Marshal.IsComObject(obj) Then
            ' Balance each COM reference obtained above. Do not FinalRelease shared RCWs:
            ' Excel can return the same RCW for several references to one sheet/collection.
            System.Runtime.InteropServices.Marshal.ReleaseComObject(obj)
        End If
    End Sub

Dim openBook As System.Func(Of Object, String, Boolean, Object) =
    Function(books As Object, path As String, openReadOnly As Boolean) As Object
        ' Open without updating external links, with macros disabled at Application level.
        Return xl(books, "Open", Microsoft.VisualBasic.CallType.Method,
            New Object() {path, 0, openReadOnly, System.Type.Missing, "", "", True,
                          System.Type.Missing, System.Type.Missing, False, False,
                          System.Type.Missing, False, True, 0})
    End Function

Dim sheetNames As System.Func(Of Object, System.Collections.Generic.HashSet(Of String)) =
    Function(book As Object) As System.Collections.Generic.HashSet(Of String)
        Dim names As New System.Collections.Generic.HashSet(Of String)(System.StringComparer.OrdinalIgnoreCase)
        Dim sheets As Object = Nothing
        Try
            sheets = xl(book, "Worksheets", Microsoft.VisualBasic.CallType.Get, New Object() {})
            Dim count As Integer = System.Convert.ToInt32(xl(sheets, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            For index As Integer = 1 To count
                Dim sheet As Object = Nothing
                Try
                    sheet = xl(sheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {index})
                    names.Add(System.Convert.ToString(xl(sheet, "Name", Microsoft.VisualBasic.CallType.Get, New Object() {})))
                Finally
                    releaseCom(sheet)
                End Try
            Next
        Finally
            releaseCom(sheets)
        End Try
        Return names
    End Function

Dim hashStream As System.Func(Of System.IO.Stream, String) =
    Function(stream As System.IO.Stream) As String
        stream.Position = 0
        Using sha As System.Security.Cryptography.SHA256 = System.Security.Cryptography.SHA256.Create()
            Return System.Convert.ToBase64String(sha.ComputeHash(stream))
        End Using
    End Function

' Allow other readers, but reject a file held for writing by Excel or another process.
' This checks readiness; it cannot close a WorkbookApplication owned by the caller.
Dim waitForInput As System.Action(Of String) =
    Sub(lockedInputPath As String)
        Dim timer As System.Diagnostics.Stopwatch = System.Diagnostics.Stopwatch.StartNew()
        Dim loggedWait As Boolean = False
        Do
            Try
                Using probe As New System.IO.FileStream(lockedInputPath, System.IO.FileMode.Open,
                            System.IO.FileAccess.Read, System.IO.FileShare.Read)
                    If probe.Length = 0 Then
                        Throw New System.IO.InvalidDataException("Input workbook is empty: " & System.IO.Path.GetFileName(lockedInputPath))
                    End If
                End Using
                If loggedWait Then log("INPUT_READY | File=" & System.IO.Path.GetFileName(lockedInputPath))
                Exit Do
            Catch ex As System.IO.IOException
                Dim nativeError As Integer = ex.HResult And &HFFFF
                If nativeError <> 32 AndAlso nativeError <> 33 Then Throw
                Dim remainingMilliseconds As Double =
                    CDbl(inputWaitTimeoutSeconds) * 1000.0 - CDbl(timer.ElapsedMilliseconds)
                If remainingMilliseconds <= 0 Then
                    ' No InnerException: the caller's root-exception loop must retain this instruction.
                    Throw New System.IO.IOException(
                        "Input workbook is still locked after " & inputWaitTimeoutSeconds.ToString() &
                        " seconds: " & lockedInputPath &
                        ". Save and close the workbook in the old process. Run aggregation outside every Excel Application Scope that uses this file.")
                End If
                If Not loggedWait Then
                    log("WAITING_FOR_INPUT | File=" & System.IO.Path.GetFileName(lockedInputPath) &
                        " | TimeoutSeconds=" & inputWaitTimeoutSeconds.ToString())
                    loggedWait = True
                End If
                System.Threading.Thread.Sleep(CInt(System.Math.Min(CDbl(inputPollMilliseconds),
                    System.Math.Ceiling(remainingMilliseconds))))
            End Try
        Loop
    End Sub

' The keeper must belong to this workbook and stay visible before deleting any sheet.
Dim deleteSheetSafely As System.Action(Of Object, Object, Object) =
    Sub(book As Object, sheetToDelete As Object, keeper As Object)
        Dim keeperParent As Object = Nothing
        Dim deleteParent As Object = Nothing
        Try
            Dim bookName As String = System.Convert.ToString(xl(book, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            keeperParent = xl(keeper, "Parent", Microsoft.VisualBasic.CallType.Get, New Object() {})
            deleteParent = xl(sheetToDelete, "Parent", Microsoft.VisualBasic.CallType.Get, New Object() {})
            If Not String.Equals(bookName,
                    System.Convert.ToString(xl(keeperParent, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {})),
                    System.StringComparison.OrdinalIgnoreCase) OrElse
               Not String.Equals(bookName,
                    System.Convert.ToString(xl(deleteParent, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {})),
                    System.StringComparison.OrdinalIgnoreCase) Then
                Throw New System.InvalidOperationException("Sheet deletion references the wrong workbook.")
            End If
            Dim keeperName As String = System.Convert.ToString(xl(keeper, "Name", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            Dim deleteName As String = System.Convert.ToString(xl(sheetToDelete, "Name", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            If String.Equals(keeperName, deleteName, System.StringComparison.OrdinalIgnoreCase) Then
                Throw New System.InvalidOperationException("The copied report sheet must not be deleted.")
            End If
            xl(keeper, "Visible", Microsoft.VisualBasic.CallType.Let, New Object() {-1})
            If System.Convert.ToInt32(xl(keeper, "Visible", Microsoft.VisualBasic.CallType.Get, New Object() {})) <> -1 Then
                Throw New System.InvalidOperationException("The copied report sheet is not visible before deletion.")
            End If
            xl(sheetToDelete, "Delete", Microsoft.VisualBasic.CallType.Method, New Object() {})
        Finally
            releaseCom(deleteParent)
            releaseCom(keeperParent)
        End Try
    End Sub

            Dim activeCopyBook As Object = Nothing
            Dim activeCopyBookName As String = String.Empty
            Dim booksAfterCopy As Integer = System.Convert.ToInt32(xl(workbooks, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            currentStep = "Capture workbook after copy"
            Try
                activeCopyBook = xl(excel, "ActiveWorkbook", Microsoft.VisualBasic.CallType.Get, New Object() {})
                If activeCopyBook IsNot Nothing Then
                    activeCopyBookName = System.Convert.ToString(xl(activeCopyBook, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {}))
                End If
                If Not existed Then
                    If activeCopyBook Is Nothing OrElse
                       String.Equals(activeCopyBookName, sourcePath, System.StringComparison.OrdinalIgnoreCase) OrElse
                       String.Equals(activeCopyBookName, groupsPath, System.StringComparison.OrdinalIgnoreCase) Then
                        Throw New System.InvalidOperationException("Copy did not return a new workbook. ActiveWorkbook=" & activeCopyBookName &
                            "; WorkbooksBefore=" & booksBeforeCopy.ToString() & "; WorkbooksAfter=" & booksAfterCopy.ToString())
                    End If
                    destinationBook = activeCopyBook
                    activeCopyBook = Nothing
                End If
            Finally
                releaseCom(activeCopyBook)
            End Try

            currentStep = "Verify copied worksheet in destination"
            destinationSheets = xl(destinationBook, "Worksheets", Microsoft.VisualBasic.CallType.Get, New Object() {})
            Dim worksheetsAfterCopy As Integer = System.Convert.ToInt32(xl(destinationSheets, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            Dim namesAfterCopy As System.Collections.Generic.HashSet(Of String) = sheetNames(destinationBook)
            Dim addedSheetNames As New System.Collections.Generic.List(Of String)()
            For Each copiedName As String In namesAfterCopy
                If Not namesBeforeCopy.Contains(copiedName) Then addedSheetNames.Add(copiedName)
            Next
            Dim destinationBookName As String = System.Convert.ToString(xl(destinationBook, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            Dim copyDetails As String = "Group=" & groupName & " | Mode=" & copyMode &
                " | Destination=" & destinationBookName & " | ActiveWorkbook=" & activeCopyBookName &
                " | BeforeCount=" & worksheetsBeforeCopy.ToString() & " | AfterCount=" & worksheetsAfterCopy.ToString() &
                " | BeforeNameCount=" & namesBeforeCopy.Count.ToString() & " | AfterNameCount=" & namesAfterCopy.Count.ToString() &
                " | AddedCount=" & addedSheetNames.Count.ToString() &
                " | WorkbooksBefore=" & booksBeforeCopy.ToString() & " | WorkbooksAfter=" & booksAfterCopy.ToString() &
                " | BeforeSheets=" & String.Join(", ", (New System.Collections.Generic.List(Of String)(namesBeforeCopy)).ToArray()) &
                " | AfterSheets=" & String.Join(", ", (New System.Collections.Generic.List(Of String)(namesAfterCopy)).ToArray())
            log("COPY_RESULT | " & copyDetails)
            If worksheetsAfterCopy <> worksheetsBeforeCopy + 1 OrElse
               namesBeforeCopy.Count <> worksheetsBeforeCopy OrElse namesAfterCopy.Count <> worksheetsAfterCopy OrElse
               addedSheetNames.Count <> 1 OrElse
               booksAfterCopy <> booksBeforeCopy + If(existed, 0, 1) Then
                Throw New System.InvalidOperationException("Copy verification failed. " & copyDetails)
            End If
            If Not existed AndAlso Not String.Equals(addedSheetNames(0), groupName, System.StringComparison.OrdinalIgnoreCase) Then
                Throw New System.InvalidOperationException("New workbook contains the wrong group sheet. " & copyDetails)
            End If
            For Each beforeName As String In namesBeforeCopy
                If Not namesAfterCopy.Contains(beforeName) Then
                    Throw New System.InvalidOperationException("Copy removed an existing destination sheet. " & copyDetails)
                End If
            Next
            If System.Convert.ToBoolean(xl(destinationBook, "ReadOnly", Microsoft.VisualBasic.CallType.Get, New Object() {})) Then
                Throw New System.UnauthorizedAccessException("Staged output workbook was opened read-only: " & fileName)
            End If
            copiedSheet = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {addedSheetNames(0)})
            Dim copiedParent As Object = Nothing
            Try
                copiedParent = xl(copiedSheet, "Parent", Microsoft.VisualBasic.CallType.Get, New Object() {})
                If Not String.Equals(destinationBookName,
                       System.Convert.ToString(xl(copiedParent, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {})),
                       System.StringComparison.OrdinalIgnoreCase) Then
                    Throw New System.InvalidOperationException("Copied sheet belongs to the wrong workbook. " & copyDetails)
                End If
            Finally
                releaseCom(copiedParent)
            End Try
            xl(copiedSheet, "Visible", Microsoft.VisualBasic.CallType.Let, New Object() {-1}) ' xlSheetVisible
            If System.Convert.ToInt32(xl(copiedSheet, "Visible", Microsoft.VisualBasic.CallType.Get, New Object() {})) <> -1 Then
                Throw New System.InvalidOperationException("Copied destination sheet could not be made visible: " & groupName)
            End If

            currentStep = "Freeze copied formulas if requested"
            If in_ValuesOnly Then
                ' Freeze the copied cells using Excel's PasteSpecial; no Select/Activate is used.
                usedRange = xl(copiedSheet, "UsedRange", Microsoft.VisualBasic.CallType.Get, New Object() {})
                Dim formulaState As Object = xl(usedRange, "HasFormula", Microsoft.VisualBasic.CallType.Get, New Object() {})
                Dim noFormulas As Boolean = TypeOf formulaState Is Boolean AndAlso Not System.Convert.ToBoolean(formulaState)
                If Not noFormulas Then
                    If System.Convert.ToBoolean(xl(copiedSheet, "FilterMode", Microsoft.VisualBasic.CallType.Get, New Object() {})) Then
                        Throw New System.IO.InvalidDataException("Cannot freeze formulas while an AutoFilter is actively hiding rows. Finalize the group data first: " & groupName)
                    End If
                    xl(usedRange, "Copy", Microsoft.VisualBasic.CallType.Method, New Object() {})
                    xl(usedRange, "PasteSpecial", Microsoft.VisualBasic.CallType.Method, New Object() {-4163, -4142, False, False}) ' xlPasteValues, xlNone
                    xl(excel, "CutCopyMode", Microsoft.VisualBasic.CallType.Let, New Object() {False})
                End If
            End If
            If previousNames.Contains(reportName) Then
                currentStep = "Replace existing report sheet"
                oldSheet = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {reportName})
                deleteSheetSafely(destinationBook, oldSheet, copiedSheet)
            End If
            currentStep = "Name copied report sheet"
            xl(copiedSheet, "Name", Microsoft.VisualBasic.CallType.Let, New Object() {reportName})

            ' Report tabs remain ordered 1,2,3,4, including an out-of-order recovery run.
            currentStep = "Order report sheets"
            Dim namesNow As System.Collections.Generic.HashSet(Of String) = sheetNames(destinationBook)
            For number As Integer = 4 To 1 Step -1
                Dim name As String = "Report " & number.ToString()
                If Not namesNow.Contains(name) Then Continue For
                Dim firstSheet As Object = Nothing
                Dim movingSheet As Object = Nothing
                Try
                    firstSheet = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {1})
                    If String.Equals(System.Convert.ToString(xl(firstSheet, "Name", Microsoft.VisualBasic.CallType.Get, New Object() {})), name,
                                     System.StringComparison.OrdinalIgnoreCase) Then Continue For
                    movingSheet = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {name})
                    xlMethod(movingSheet, "Move", New Object() {firstSheet}, New String() {"Before"})
                Finally
                    releaseCom(movingSheet)
                    releaseCom(firstSheet)
                End Try
            Next
            Dim resultNames As System.Collections.Generic.HashSet(Of String) = sheetNames(destinationBook)
            If Not resultNames.Contains(reportName) Then
                Throw New System.InvalidOperationException("Copied report sheet is missing before save: " & fileName)
            End If
            For Each previousName As String In previousNames
                If Not resultNames.Contains(previousName) Then
                    Throw New System.InvalidOperationException("An existing sheet would be lost: " & previousName)
                End If
            Next

            currentStep = "Save staged group workbook"
            If existed Then
                xl(destinationBook, "Save", Microsoft.VisualBasic.CallType.Method, New Object() {})
            Else
                xl(destinationBook, "SaveAs", Microsoft.VisualBasic.CallType.Method, New Object() {stagedPath, 51}) ' xlOpenXMLWorkbook .xlsx
            End If
            xl(destinationBook, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
            releaseCom(destinationBook)
            destinationBook = Nothing
            If Not System.IO.File.Exists(stagedPath) OrElse (New System.IO.FileInfo(stagedPath)).Length = 0 Then
                Throw New System.IO.IOException("Excel did not create a nonempty staged workbook: " & fileName)
            End If
            log("STAGED | Group=" & groupName & " | Sheet=" & reportName)
        Finally
            If restoreSourceVisibility AndAlso sourceSheet IsNot Nothing Then
                Try
                    ' Source is read-only and always closed without saving.
                    xl(sourceSheet, "Visible", Microsoft.VisualBasic.CallType.Let, New Object() {originalSourceVisibility})
                Catch
                    ' Source file is never saved; do not replace the original export error.
                End Try
            End If
            releaseCom(usedRange)
            releaseCom(oldSheet)
            releaseCom(copiedSheet)
            releaseCom(anchorSheet)
            releaseCom(sourceSheet)
            releaseCom(sourceSheets)
            releaseCom(destinationSheets)
        End Try
    Next

    ' Close our workbooks and Excel before moving/replacing physical files.
    currentStep = "Close export Excel instance"
    xl(sourceBook, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
    releaseCom(sourceBook)
    sourceBook = Nothing
    xl(groupsBook, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
    releaseCom(groupsBook)
    groupsBook = Nothing
    xl(excel, "Quit", Microsoft.VisualBasic.CallType.Method, New Object() {})
    releaseCom(workbooks)
    workbooks = Nothing
    releaseCom(excel)
    excel = Nothing

    ' Publication is per file, not one transaction across every group.
    ' If a later file fails, rerunning safely replaces this report's sheet again.
    currentStep = "Publish group workbooks"
    For Each groupName As String In groups
        currentGroup = groupName
        Dim fileName As String = "New Report - " & groupName & ".xlsx"
        Dim finalPath As String = System.IO.Path.Combine(periodFolder, fileName)
        Dim stagedPath As String = System.IO.Path.Combine(stageFolder, fileName)
        Dim originalHash As String = expectedHashes(finalPath)
        If originalHash Is Nothing Then
            ' Move does not overwrite a file that unexpectedly appeared.
            System.IO.File.Move(stagedPath, finalPath)
        Else
            If Not System.IO.File.Exists(finalPath) Then
                Throw New System.IO.IOException("Output workbook disappeared during export: " & fileName)
            End If
            Using guard As New System.IO.FileStream(finalPath, System.IO.FileMode.Open,
                        System.IO.FileAccess.ReadWrite, System.IO.FileShare.None)
                If Not String.Equals(hashStream(guard), originalHash, System.StringComparison.Ordinal) Then
                    Throw New System.IO.IOException("Output workbook changed during export. Rerun after resolving the concurrent edit: " & fileName)
                End If
            End Using
            System.IO.Directory.CreateDirectory(backupFolder)
            ' No Delete + overwrite fallback. If Replace is unsupported on the share, fail safely.
            System.IO.File.Replace(stagedPath, finalPath, System.IO.Path.Combine(backupFolder, fileName), False)
        End If
        completedFiles.Add(finalPath)
        log("COMMITTED | Group=" & groupName & " | Sheet=" & reportName)
    Next

    out_UpdatedFiles = completedFiles.ToArray()
    out_Summary = "RunId=" & runId & " | Period=" & reportMonth & " | InputPeriod=" & periodInput & " | " & reportName &
                  " | UpdatedFiles=" & completedFiles.Count.ToString() &
                  " | DuplicateNamesIgnored=" & duplicateCount.ToString()
    log("SUCCESS | UpdatedFiles=" & completedFiles.Count.ToString())
    success = True
Catch ex As System.Exception
    Try
        If System.IO.Directory.Exists(journalFolder) Then
            log("FAILED | Group=" & currentGroup & " | Step=" & currentStep & " | Type=" & ex.GetType().FullName &
                " | Message=" & ex.Message.Replace(System.Environment.NewLine, " ") &
                " | CommittedFiles=" & completedFiles.Count.ToString())
        End If
    Catch
        ' Do not hide the original failure if writing the failure log also fails.
    End Try
    Throw
Finally
    ' Cleanup uses only objects created here. Never Kill Process all EXCEL instances.
    For Each book As Object In New Object() {destinationBook, sourceBook, groupsBook}
        If book Is Nothing Then Continue For
        Try
            xl(book, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
        Catch
        Finally
            Try
                releaseCom(book)
            Catch
            End Try
        End Try
    Next
    If excel IsNot Nothing Then
        Try
            xl(excel, "Quit", Microsoft.VisualBasic.CallType.Method, New Object() {})
        Catch
        End Try
    End If
    Try
        releaseCom(workbooks)
    Catch
    End Try
    Try
        releaseCom(excel)
    Catch
    End Try
    If success Then
        Try
            System.IO.Directory.Delete(stageFolder, True)
        Catch
            ' Final files are already published; leftover staging can be removed later.
        End Try
    End If
    If lockStream IsNot Nothing Then lockStream.Dispose()
End Try
