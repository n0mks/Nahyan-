' 3. EXCEL COM ACCESS
' Centralize retries. Retry only calls Excel explicitly rejected as busy;
' file errors, invalid data and other COM failures must reach the outer Catch.
Dim invokeCom As System.Func(Of System.Func(Of Object), Object) =
    Function(operation As System.Func(Of Object)) As Object
        For attempt As Integer = 1 To 3
            Try
                Return operation()
            Catch ex As System.Exception
                Dim cause As System.Exception = ex
                While TypeOf cause Is System.Reflection.TargetInvocationException AndAlso cause.InnerException IsNot Nothing
                    cause = cause.InnerException
                End While
                Dim busy As System.Runtime.InteropServices.COMException = TryCast(cause, System.Runtime.InteropServices.COMException)
                If busy Is Nothing OrElse (busy.ErrorCode <> -2147418111 AndAlso busy.ErrorCode <> -2147417846) OrElse attempt = 3 Then Throw
                System.Threading.Thread.Sleep(250 * attempt)
            End Try
        Next
        Throw New System.InvalidOperationException("Excel COM retry limit reached.")
    End Function

' CallByName preserves compatibility with Option Strict On and Windows-Legacy.
Dim xl As System.Func(Of Object, String, Microsoft.VisualBasic.CallType, Object(), Object) =
    Function(target As Object, member As String, kind As Microsoft.VisualBasic.CallType, args As Object()) As Object
        Return invokeCom(Function() Microsoft.VisualBasic.Interaction.CallByName(target, member, kind, args))
    End Function
' Add, Move and Range.Copy need explicit argument names to bind the destination.
Dim xlMethod As System.Func(Of Object, String, Object(), String(), Object) =
    Function(target As Object, member As String, args As Object(), names As String()) As Object
        Return invokeCom(Function() target.GetType().InvokeMember(member,
            System.Reflection.BindingFlags.InvokeMethod Or System.Reflection.BindingFlags.Public Or
            System.Reflection.BindingFlags.Instance Or System.Reflection.BindingFlags.OptionalParamBinding,
            Nothing, target, args, Nothing, System.Globalization.CultureInfo.GetCultureInfo("en-US"), names))
    End Function

' Small wrappers keep the workflow readable without changing the COM calls.
Dim readProp As System.Func(Of Object, String, Object) =
    Function(obj As Object, name As String) xl(obj, name, Microsoft.VisualBasic.CallType.Get, New Object() {})
Dim itemAt As System.Func(Of Object, Object, Object) =
    Function(collection As Object, index As Object) xl(collection, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {index})
Dim rangeAt As System.Func(Of Object, String, Object) =
    Function(sheet As Object, address As String) xl(sheet, "Range", Microsoft.VisualBasic.CallType.Get, New Object() {address})
Dim writeProp As System.Action(Of Object, String, Object) =
    Sub(obj As Object, name As String, value As Object)
        xl(obj, name, Microsoft.VisualBasic.CallType.Let, New Object() {value})
    End Sub
Dim runMethod As System.Action(Of Object, String) =
    Sub(obj As Object, name As String)
        xl(obj, name, Microsoft.VisualBasic.CallType.Method, New Object() {})
    End Sub
Dim closeBook As System.Action(Of Object) =
    Sub(book As Object)
        xl(book, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
    End Sub

Dim releaseCom As System.Action(Of Object) =
    Sub(obj As Object)
        If obj IsNot Nothing AndAlso System.Runtime.InteropServices.Marshal.IsComObject(obj) Then
            ' Balance each COM reference obtained above. Do not FinalRelease shared RCWs:
            ' Excel can return the same RCW for several references to one sheet/collection.
            System.Runtime.InteropServices.Marshal.ReleaseComObject(obj)
        End If
    End Sub

' Balance references in the supplied order, including repeated RCWs returned by COM.
Dim releaseAll As System.Action(Of Object()) =
    Sub(objects As Object())
        For Each obj As Object In objects
            releaseCom(obj)
        Next
    End Sub

' Reuse one loop for both column widths and row heights; preserve hidden state.
Dim copyLayout As System.Action(Of Object, Object, String, String, Integer) =
    Sub(source As Object, target As Object, axis As String, sizeProperty As String, count As Integer)
        Dim sourceAxis As Object = Nothing, targetAxis As Object = Nothing
        Try
            sourceAxis = readProp(source, axis)
            targetAxis = readProp(target, axis)
            For index As Integer = 1 To count
                Dim sourceItem As Object = Nothing, targetItem As Object = Nothing
                Try
                    sourceItem = itemAt(sourceAxis, index)
                    targetItem = itemAt(targetAxis, index)
                    writeProp(targetItem, sizeProperty, readProp(sourceItem, sizeProperty))
                    writeProp(targetItem, "Hidden", readProp(sourceItem, "Hidden"))
                Finally
                    releaseAll(New Object() {targetItem, sourceItem})
                End Try
            Next
        Finally
            releaseAll(New Object() {targetAxis, sourceAxis})
        End Try
    End Sub

Dim openBook As System.Func(Of Object, String, Boolean, Object) =
    Function(books As Object, path As String, openReadOnly As Boolean) As Object
        ' Open without updating external links, with macros disabled at Application level.
        Return xl(books, "Open", Microsoft.VisualBasic.CallType.Method, New Object() {path, 0, openReadOnly, System.Type.Missing, "", "", True,
                          System.Type.Missing, System.Type.Missing, False, False,
                          System.Type.Missing, False, True, 0})
    End Function

Dim sheetNames As System.Func(Of Object, System.Collections.Generic.HashSet(Of String)) =
    Function(book As Object) As System.Collections.Generic.HashSet(Of String)
        Dim names As New System.Collections.Generic.HashSet(Of String)(nameComparer)
        Dim sheets As Object = Nothing
        Try
            sheets = readProp(book, "Worksheets")
            Dim count As Integer = System.Convert.ToInt32(readProp(sheets, "Count"))
            For index As Integer = 1 To count
                Dim sheet As Object = Nothing
                Try
                    sheet = itemAt(sheets, index)
                    names.Add(System.Convert.ToString(readProp(sheet, "Name")))
                Finally
                    releaseCom(sheet)
                End Try
            Next
        Finally
            releaseCom(sheets)
        End Try
        Return names
    End Function

' 4. CELL DATA HELPERS — support scalar cells and Excel's 1-based SAFEARRAY values.
Dim columnName As System.Func(Of Integer, String) =
    Function(columnNumber As Integer) As String
        Dim name As String = String.Empty
        Do While columnNumber > 0
            columnNumber -= 1
            name = System.Convert.ToChar(65 + (columnNumber Mod 26)).ToString() & name
            columnNumber = columnNumber \ 26
        Loop
        Return name
    End Function

Dim valueAt As System.Func(Of Object, Integer, Integer, Object) =
    Function(raw As Object, rowOffset As Integer, columnOffset As Integer) As Object
        Dim matrix As System.Array = TryCast(raw, System.Array)
        If matrix IsNot Nothing Then
            If matrix.Rank <> 2 Then Throw New System.InvalidOperationException("Excel returned an unexpected value array.")
            Return matrix.GetValue(matrix.GetLowerBound(0) + rowOffset, matrix.GetLowerBound(1) + columnOffset)
        End If
        If rowOffset = 0 AndAlso columnOffset = 0 Then Return raw
        Return Nothing
    End Function

Dim hasValue As System.Func(Of Object, Boolean) =
    Function(value As Object) As Boolean
        If value Is Nothing OrElse value Is System.DBNull.Value Then Return False
        If TypeOf value Is String Then Return DirectCast(value, String).Length > 0
        Return True
    End Function

' A row is populated when any cell has a value; numeric zero is not a blank cell.
Dim populatedRow As System.Func(Of Object, Integer, Integer, Boolean) =
    Function(values As Object, rowOffset As Integer, columnCount As Integer) As Boolean
        For columnOffset As Integer = 0 To columnCount - 1
            If hasValue(valueAt(values, rowOffset, columnOffset)) Then Return True
        Next
        Return False
    End Function

' Keep zero distinct from blank, and compare Excel error values by their error code.
Dim valuesMatch As System.Func(Of Object, Object, Boolean) =
    Function(left As Object, right As Object) As Boolean
        If Not hasValue(left) AndAlso Not hasValue(right) Then Return True
        Dim leftError As System.Runtime.InteropServices.ErrorWrapper = TryCast(left, System.Runtime.InteropServices.ErrorWrapper)
        Dim rightError As System.Runtime.InteropServices.ErrorWrapper = TryCast(right, System.Runtime.InteropServices.ErrorWrapper)
        If leftError IsNot Nothing OrElse rightError IsNot Nothing Then
            Return leftError IsNot Nothing AndAlso rightError IsNot Nothing AndAlso leftError.ErrorCode = rightError.ErrorCode
        End If
        Return System.Object.Equals(left, right)
    End Function

' Select a header template from this report only; each process may have different columns.
Dim headerInfo As System.Func(Of Object, Integer()) =
    Function(sheet As Object) As Integer()
        Dim area As Object = Nothing, areaColumns As Object = Nothing, headers As Object = Nothing
        Try
            area = readProp(sheet, "UsedRange")
            areaColumns = readProp(area, "Columns")
            Dim lastColumn As Integer = System.Convert.ToInt32(readProp(area, "Column")) +
                System.Convert.ToInt32(readProp(areaColumns, "Count")) - 1
            headers = rangeAt(sheet, "A" & headerRow.ToString() & ":" & columnName(lastColumn) & headerRow.ToString())
            Dim raw As Object = readProp(headers, "Value2")
            Dim labelCount As Integer = 0
            For columnOffset As Integer = 0 To lastColumn - 1
                Dim label As Object = valueAt(raw, 0, columnOffset)
                If TypeOf label Is String AndAlso Not String.IsNullOrWhiteSpace(DirectCast(label, String)) Then labelCount += 1
            Next
            Return New Integer() {labelCount, lastColumn}
        Finally
            releaseAll(New Object() {headers, areaColumns, area})
        End Try
    End Function

Dim hashStream As System.Func(Of System.IO.Stream, String) =
    Function(stream As System.IO.Stream) As String
        stream.Position = 0
        Using sha As System.Security.Cryptography.SHA256 = System.Security.Cryptography.SHA256.Create()
            Return System.Convert.ToBase64String(sha.ComputeHash(stream))
        End Using
    End Function

' Allow other readers, but reject a file held for writing by Excel or another process.
' This checks readiness; it cannot close a WorkbookApplication owned by the caller.
Dim waitForInput As System.Action(Of String) =
    Sub(lockedInputPath As String)
        Dim timer As System.Diagnostics.Stopwatch = System.Diagnostics.Stopwatch.StartNew()
        Dim loggedWait As Boolean = False
        Do
            Try
                Using probe As New System.IO.FileStream(lockedInputPath, System.IO.FileMode.Open, System.IO.FileAccess.Read, System.IO.FileShare.Read)
                    If probe.Length = 0 Then
                        Throw New System.IO.InvalidDataException("Input workbook is empty: " & System.IO.Path.GetFileName(lockedInputPath))
                    End If
                End Using
                If loggedWait Then log("INPUT_READY | File=" & System.IO.Path.GetFileName(lockedInputPath))
                Exit Do
            Catch ex As System.IO.IOException
                Dim nativeError As Integer = ex.HResult And &HFFFF
                If nativeError <> 32 AndAlso nativeError <> 33 Then Throw
                Dim remainingMilliseconds As Double =
                    CDbl(inputWaitTimeoutSeconds) * 1000.0 - CDbl(timer.ElapsedMilliseconds)
                If remainingMilliseconds <= 0 Then
                    ' No InnerException: the caller's root-exception loop must retain this instruction.
                    Throw New System.IO.IOException("Input workbook is still locked after " & inputWaitTimeoutSeconds.ToString() &
                        " seconds: " & lockedInputPath &
                        ". Save and close the workbook in the old process. Run aggregation outside every Excel Application Scope that uses this file.")
                End If
                If Not loggedWait Then
                    log("WAITING_FOR_INPUT | File=" & System.IO.Path.GetFileName(lockedInputPath) &
                        " | TimeoutSeconds=" & inputWaitTimeoutSeconds.ToString())
                    loggedWait = True
                End If
                System.Threading.Thread.Sleep(CInt(System.Math.Min(CDbl(inputPollMilliseconds), System.Math.Ceiling(remainingMilliseconds))))
            End Try
        Loop
    End Sub

' The keeper must belong to this workbook and stay visible before deleting any sheet.
Dim deleteSheetSafely As System.Action(Of Object, Object, Object) =
    Sub(book As Object, sheetToDelete As Object, keeper As Object)
        Dim keeperParent As Object = Nothing, deleteParent As Object = Nothing
        Try
            Dim bookName As String = System.Convert.ToString(readProp(book, "FullName"))
            keeperParent = readProp(keeper, "Parent")
            deleteParent = readProp(sheetToDelete, "Parent")
            If Not String.Equals(bookName, System.Convert.ToString(readProp(keeperParent, "FullName")), ignoreCase) OrElse
               Not String.Equals(bookName, System.Convert.ToString(readProp(deleteParent, "FullName")), ignoreCase) Then
                Throw New System.InvalidOperationException("Sheet deletion references the wrong workbook.")
            End If
            Dim keeperName As String = System.Convert.ToString(readProp(keeper, "Name"))
            Dim deleteName As String = System.Convert.ToString(readProp(sheetToDelete, "Name"))
            If String.Equals(keeperName, deleteName, ignoreCase) Then
                Throw New System.InvalidOperationException("The copied report sheet must not be deleted.")
            End If
            writeProp(keeper, "Visible", -1)
            If System.Convert.ToInt32(readProp(keeper, "Visible")) <> -1 Then
                Throw New System.InvalidOperationException("The copied report sheet is not visible before deletion.")
            End If
            runMethod(sheetToDelete, "Delete")
        Finally
            releaseAll(New Object() {deleteParent, keeperParent})
        End Try
    End Sub

' 6. START THE EXPORT — a monthly handle lock prevents overlapping workflow runs.
Try
    ' Keep this handle open. A leftover lock file is harmless after its handle is closed.
    ' Do not delete the lock file: the open handle is what provides exclusion.
    currentStep = "Acquire monthly output lock"
    Try
        lockStream = New System.IO.FileStream(System.IO.Path.Combine(periodFolder, ".monthly-report-aggregation.lock"),
            System.IO.FileMode.OpenOrCreate, System.IO.FileAccess.ReadWrite, System.IO.FileShare.None)
    Catch ex As System.IO.IOException
        Throw New System.IO.IOException("Cannot acquire the monthly export lock. Another export may be running, or the folder may be unavailable.", ex)
    End Try

    currentStep = "Create export work folders"
    System.IO.Directory.CreateDirectory(stageFolder)
    System.IO.Directory.CreateDirectory(journalFolder)
    log("START | Build=2026-10-13-RangeExport3 | Revision=Compact4 | RunId=" & runId & " | Period=" & reportMonth & " | InputPeriod=" & periodInput & " | Report=" & in_ReportNumber.ToString())

    currentStep = "Wait for source workbook"
    waitForInput(sourcePath)
    currentStep = "Wait for CRM Group workbook"
    waitForInput(groupsPath)

    currentStep = "Start export Excel instance"
    Dim excelType As System.Type = System.Type.GetTypeFromProgID("Excel.Application")
    If excelType Is Nothing Then Throw New System.InvalidOperationException("Microsoft Excel is not installed or its COM registration is unavailable.")
    excel = System.Activator.CreateInstance(excelType)
    writeProp(excel, "Visible", False)
    writeProp(excel, "DisplayAlerts", False)
    writeProp(excel, "EnableEvents", False)
    writeProp(excel, "ScreenUpdating", False)
    writeProp(excel, "AskToUpdateLinks", False)
    writeProp(excel, "AutomationSecurity", 3) ' ForceDisable
    workbooks = readProp(excel, "Workbooks")
    currentStep = "Open CRM Group workbook"
    groupsBook = openBook(workbooks, groupsPath, True)
    ' Set Calculation after a workbook exists. Use the saved, finalized report values.
    writeProp(excel, "Calculation", -4135) ' xlCalculationManual
    writeProp(excel, "CalculateBeforeSave", False)

    ' 7. READ CRM GROUPS FROM COLUMN A, STARTING AT A2
    ' Preserve CRM order, ignore blank/duplicate names and reject unusable sheet/file names.
    Dim groupSheets As Object = Nothing, groupSheet As Object = Nothing, rows As Object = Nothing, cells As Object = Nothing
    Dim bottomCell As Object = Nothing, lastCell As Object = Nothing, listRange As Object = Nothing
    currentStep = "Read CRM group list"
    Try
        groupSheets = readProp(groupsBook, "Worksheets")
        If String.IsNullOrWhiteSpace(in_CrmGroupSheetName) Then
            groupSheet = itemAt(groupSheets, 1)
            log("INFO | CRM Group worksheet=first worksheet")
        Else
            If Not sheetNames(groupsBook).Contains(in_CrmGroupSheetName) Then
                Throw New System.IO.InvalidDataException("CRM Group worksheet was not found: " & in_CrmGroupSheetName)
            End If
            groupSheet = itemAt(groupSheets, in_CrmGroupSheetName)
        End If

        rows = readProp(groupSheet, "Rows")
        cells = readProp(groupSheet, "Cells")
        Dim maxRow As Integer = System.Convert.ToInt32(readProp(rows, "Count"))
        bottomCell = xl(cells, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {maxRow, 1})
        lastCell = xl(bottomCell, "End", Microsoft.VisualBasic.CallType.Get, New Object() {-4162}) ' xlUp
        Dim lastRow As Integer = System.Convert.ToInt32(readProp(lastCell, "Row"))
        If lastRow < 2 Then Throw New System.IO.InvalidDataException("CRM Group column A contains no groups from A2 onward.")
        listRange = rangeAt(groupSheet, "A2:A" & lastRow.ToString())
        Dim raw As Object = readProp(listRange, "Value2")
        Dim values As System.Array = TryCast(raw, System.Array)
        Dim seen As New System.Collections.Generic.HashSet(Of String)(nameComparer)
        Dim invalidChars As Char() = ("<>:""/\|?*[]").ToCharArray()
        For offset As Integer = 0 To lastRow - 2
            Dim value As Object = raw
            If values IsNot Nothing Then
                value = values.GetValue(values.GetLowerBound(0) + offset, values.GetLowerBound(1))
            End If
            If value Is Nothing Then Continue For
            If Not TypeOf value Is String Then
                Throw New System.IO.InvalidDataException("CRM Group A" & (offset + 2).ToString() & " must contain a text group name.")
            End If
            Dim groupName As String = DirectCast(value, String).Trim()
            If groupName.Length = 0 Then Continue For
            Dim invalid As Boolean = groupName.Length > 31 OrElse groupName.IndexOfAny(invalidChars) >= 0 OrElse
                                     groupName.StartsWith("'", System.StringComparison.Ordinal) OrElse
                                     groupName.EndsWith("'", System.StringComparison.Ordinal)
            For Each character As Char In groupName
                If System.Char.IsControl(character) Then invalid = True
            Next
            If invalid Then
                Throw New System.IO.InvalidDataException("Invalid group/sheet/file name in CRM Group A" & (offset + 2).ToString() & ": " & groupName)
            End If
            If seen.Add(groupName) Then
                groups.Add(groupName)
            Else
                duplicateCount += 1
            End If
        Next
        If groups.Count = 0 Then Throw New System.IO.InvalidDataException("CRM Group contains no nonblank valid groups.")
    Finally
        releaseAll(New Object() {listRange, lastCell, bottomCell, cells, rows, groupSheet, groupSheets})
    End Try

    currentStep = "Open source workbook"
    sourceBook = openBook(workbooks, sourcePath, True)
    ' 8. VALIDATE GROUP SHEETS AND CHOOSE THE HEADER TEMPLATE
    ' Missing groups use this template; sheets absent from CRM are deliberately ignored.
    currentStep = "Validate group names and select the header template"
    Dim sourceNames As System.Collections.Generic.HashSet(Of String) = sheetNames(sourceBook)
    Dim missingGroups As New System.Collections.Generic.List(Of String)()
    Dim ignoredSheets As New System.Collections.Generic.List(Of String)()
    Dim headerTemplateName As String = String.Empty
    Dim headerTemplateColumnCount As Integer = 0
    Dim bestHeaderCount As Integer = 0
    Dim sourceHeaders As New System.Collections.Generic.Dictionary(Of String, Integer())(nameComparer)
    Dim sourceDate1904 As Boolean = System.Convert.ToBoolean(readProp(sourceBook, "Date1904"))
    Dim templateSheets As Object = Nothing
    Try
        templateSheets = readProp(sourceBook, "Worksheets")
        For Each groupName As String In groups
            If Not sourceNames.Contains(groupName) Then
                missingGroups.Add(groupName)
            Else
                Dim candidate As Object = Nothing
                Try
                    candidate = itemAt(templateSheets, groupName)
                    Dim info As Integer() = headerInfo(candidate)
                    sourceHeaders.Add(groupName, info)
                    If info(0) > bestHeaderCount Then
                        bestHeaderCount = info(0)
                        headerTemplateName = groupName
                        headerTemplateColumnCount = info(1)
                    End If
                Finally
                    releaseCom(candidate)
                End Try
            End If
            Dim finalPath As String = System.IO.Path.Combine(periodFolder, "New Report - " & groupName & ".xlsx")
            If String.Equals(finalPath, sourcePath, ignoreCase) OrElse
               String.Equals(finalPath, groupsPath, ignoreCase) Then
                Throw New System.ArgumentException("An output path must not overwrite an input workbook.")
            End If
        Next
    Finally
        releaseCom(templateSheets)
    End Try
    Dim requiredGroups As New System.Collections.Generic.HashSet(Of String)(groups, nameComparer)
    For Each sourceName As String In sourceNames
        If Not requiredGroups.Contains(sourceName) Then ignoredSheets.Add(sourceName)
    Next
    If missingGroups.Count > 0 AndAlso String.Equals(missingSheetPolicy, "Fail", ignoreCase) Then
        Throw New System.IO.InvalidDataException("Source report is missing required group sheets: " & String.Join(", ", missingGroups.ToArray()))
    End If
    If Not String.Equals(missingSheetPolicy, "HeadersOnly", ignoreCase) AndAlso
       Not String.Equals(missingSheetPolicy, "Fail", ignoreCase) Then
        Throw New System.ArgumentException("missingSheetPolicy must be HeadersOnly or Fail.")
    End If
    If headerTemplateName.Length = 0 AndAlso (missingGroups.Count > 0 OrElse repairEmptyHeaderRow) Then
        Throw New System.IO.InvalidDataException("No required group sheet has a nonempty header row " & headerRow.ToString() &
            ". Supply at least one group sheet with this report's real column headers.")
    End If
    log("VALIDATED | Groups=" & groups.Count.ToString() & " | MissingSheets=" & missingGroups.Count.ToString() &
        " | HeaderTemplate=" & headerTemplateName & " | DuplicateNamesIgnored=" & duplicateCount.ToString())
    If ignoredSheets.Count > 0 Then log("IGNORED_SOURCE_SHEETS | Names=" & String.Join(", ", ignoredSheets.ToArray()))

    ' 9. BUILD EVERY GROUP FILE IN STAGING BEFORE CHANGING ANY FINAL FILE
    ' Existing outputs are copied to staging; their hashes are checked again before publication.
    For Each groupName As String In groups
        currentGroup = groupName
        currentStep = "Prepare destination workbook"
        Dim fileName As String = "New Report - " & groupName & ".xlsx"
        Dim finalPath As String = System.IO.Path.Combine(periodFolder, fileName)
        Dim stagedPath As String = System.IO.Path.Combine(stageFolder, fileName)
        Dim existed As Boolean = System.IO.File.Exists(finalPath)
        Dim missingSource As Boolean = Not sourceNames.Contains(groupName)
        Dim sourceSheetName As String = If(missingSource, headerTemplateName, groupName)
        Dim previousNames As New System.Collections.Generic.HashSet(Of String)(nameComparer)
        Dim destinationSheets As Object = Nothing, sourceSheets As Object = Nothing, sourceSheet As Object = Nothing, sourceArea As Object = Nothing
        Dim sourceAreaRows As Object = Nothing, sourceAreaColumns As Object = Nothing, sourceRange As Object = Nothing, destinationRange As Object = Nothing
        Dim anchorSheet As Object = Nothing, reportSheet As Object = Nothing, oldSheet As Object = Nothing
        Dim sourceDataRows As Integer = 0
        Try
            If existed Then
                If (System.IO.File.GetAttributes(finalPath) And System.IO.FileAttributes.ReadOnly) <> 0 Then
                    Throw New System.UnauthorizedAccessException("Output workbook is read-only: " & fileName)
                End If
                Using existing As New System.IO.FileStream(finalPath, System.IO.FileMode.Open, System.IO.FileAccess.Read, System.IO.FileShare.None)
                    expectedHashes.Add(finalPath, hashStream(existing))
                    existing.Position = 0
                    Using staged As New System.IO.FileStream(stagedPath, System.IO.FileMode.CreateNew,
                                System.IO.FileAccess.Write, System.IO.FileShare.None)
                        existing.CopyTo(staged)
                        staged.Flush(True)
                    End Using
                End Using
                destinationBook = openBook(workbooks, stagedPath, False)
                previousNames = sheetNames(destinationBook)
                If System.Convert.ToBoolean(readProp(destinationBook, "ProtectStructure")) Then
                    Throw New System.IO.InvalidDataException("Output workbook structure is protected: " & fileName)
                End If
                If System.Convert.ToBoolean(readProp(destinationBook, "Date1904")) <> sourceDate1904 Then
                    Throw New System.IO.InvalidDataException("Source and output workbook date systems differ: " & fileName)
                End If
            Else
                expectedHashes.Add(finalPath, Nothing)
                destinationBook = xl(workbooks, "Add", Microsoft.VisualBasic.CallType.Method, New Object() {-4167})
                writeProp(destinationBook, "Date1904", sourceDate1904)
            End If
            If System.Convert.ToBoolean(readProp(destinationBook, "ReadOnly")) Then
                Throw New System.UnauthorizedAccessException("Staged output is read-only: " & fileName)
            End If

            ' A. Create a known visible sheet; do not use Worksheet.Copy or ActiveWorkbook.
            currentStep = "Create an explicit report worksheet"
            destinationSheets = readProp(destinationBook, "Worksheets")
            Dim beforeCount As Integer = System.Convert.ToInt32(readProp(destinationSheets, "Count"))
            If existed Then
                runMethod(destinationBook, "Activate")
                For index As Integer = 1 To beforeCount
                    Dim candidate As Object = itemAt(destinationSheets, index)
                    Try
                        If System.Convert.ToInt32(readProp(candidate, "Visible")) = -1 Then
                            anchorSheet = candidate
                            candidate = Nothing
                            Exit For
                        End If
                    Finally
                        releaseCom(candidate)
                    End Try
                Next
                If anchorSheet Is Nothing Then Throw New System.IO.InvalidDataException("Output has no visible worksheet: " & fileName)
                reportSheet = xlMethod(destinationSheets, "Add", New Object() {anchorSheet, 1, -4167}, New String() {"Before", "Count", "Type"})
            Else
                If beforeCount <> 1 Then Throw New System.InvalidOperationException("New workbook must contain exactly one worksheet.")
                reportSheet = itemAt(destinationSheets, 1)
            End If
            If reportSheet Is Nothing Then Throw New System.InvalidOperationException("Excel did not return the report worksheet.")
            writeProp(reportSheet, "Visible", -1)
            Dim afterCount As Integer = System.Convert.ToInt32(readProp(destinationSheets, "Count"))
            If afterCount <> beforeCount + If(existed, 1, 0) Then Throw New System.InvalidOperationException("Unexpected worksheet count after creating the report sheet.")
            Dim parent As Object = Nothing
            Try
                parent = readProp(reportSheet, "Parent")
                If Not String.Equals(System.Convert.ToString(readProp(parent, "FullName")), System.Convert.ToString(readProp(destinationBook, "FullName")),
                    ignoreCase) Then
                    Throw New System.InvalidOperationException("The report worksheet belongs to the wrong workbook.")
                End If
            Finally
                releaseCom(parent)
            End Try

            ' B. Determine actual content, retaining the original title/header positions.
            ' Trim formatted trailing blanks only; zero-valued rows remain real data.
            currentStep = "Read the source group cells"
            sourceSheets = readProp(sourceBook, "Worksheets")
            sourceSheet = itemAt(sourceSheets, sourceSheetName)
            If Not missingSource AndAlso System.Convert.ToBoolean(readProp(sourceSheet, "FilterMode")) Then
                Throw New System.IO.InvalidDataException("The group sheet still has an active filter. Finalize this group's data before export: " & groupName)
            End If
            sourceArea = readProp(sourceSheet, "UsedRange")
            sourceAreaRows = readProp(sourceArea, "Rows")
            sourceAreaColumns = readProp(sourceArea, "Columns")
            Dim areaFirstRow As Integer = System.Convert.ToInt32(readProp(sourceArea, "Row"))
            Dim areaFirstColumn As Integer = System.Convert.ToInt32(readProp(sourceArea, "Column"))
            Dim areaRowCount As Integer = System.Convert.ToInt32(readProp(sourceAreaRows, "Count"))
            Dim areaColumnCount As Integer = System.Convert.ToInt32(readProp(sourceAreaColumns, "Count"))
            Dim lastColumn As Integer = areaFirstColumn + areaColumnCount - 1
            Dim lastRow As Integer = headerRow
            If missingSource Then
                lastColumn = headerTemplateColumnCount
                missingSheetCount += 1
                log("MISSING_SOURCE_SHEET | Group=" & groupName & " | Action=HeadersOnly | HeaderTemplate=" & headerTemplateName)
            Else
                Dim rawContent As Object = readProp(sourceArea, If(in_ValuesOnly, "Value2", "Formula"))
                For rowOffset As Integer = 0 To areaRowCount - 1
                    If populatedRow(rawContent, rowOffset, areaColumnCount) Then
                        lastRow = System.Math.Max(lastRow, areaFirstRow + rowOffset)
                    End If
                Next
            End If
            ' Read from A1 to keep the original title and header row positions.
            Dim address As String = "A1:" & columnName(lastColumn) & lastRow.ToString()
            sourceRange = rangeAt(sourceSheet, address)
            destinationRange = rangeAt(reportSheet, address)
            Dim sourceValues As Object = readProp(sourceRange, "Value2")
            For rowOffset As Integer = headerRow To lastRow - 1
                If populatedRow(sourceValues, rowOffset, lastColumn) Then sourceDataRows += 1
            Next

            ' C. Copy cell formatting, then optionally freeze the source's saved values.
            ' Do not recalculate here: the old process must calculate and save its report first.
            currentStep = "Transfer cells to the explicit destination range"
            log("TRANSFER_BEGIN | Group=" & groupName & " | SourceSheet=" & sourceSheetName &
                " | Range=" & address & " | DataRows=" & sourceDataRows.ToString() & " | ExistingFile=" & existed.ToString())
            xlMethod(sourceRange, "Copy", New Object() {destinationRange}, New String() {"Destination"})
            If in_ValuesOnly Then
                ' Write the source's saved values directly: no PasteSpecial or recalculation.
                writeProp(destinationRange, "Value2", sourceValues)
                currentStep = "Verify every exported cell value"
                Dim actualValues As Object = readProp(destinationRange, "Value2")
                Dim preservedTextCount As Integer = 0
                For rowOffset As Integer = 0 To lastRow - 1
                    For columnOffset As Integer = 0 To lastColumn - 1
                        Dim expectedValue As Object = valueAt(sourceValues, rowOffset, columnOffset)
                        Dim actualValue As Object = valueAt(actualValues, rowOffset, columnOffset)
                        If Not valuesMatch(expectedValue, actualValue) Then
                            ' Excel can coerce numeric-looking text under General formatting.
                            ' Keep leading zeroes and literal text, then restore its original format.
                            If TypeOf expectedValue Is String Then
                                Dim textCell As Object = Nothing
                                Try
                                    textCell = rangeAt(reportSheet, columnName(columnOffset + 1) & (rowOffset + 1).ToString())
                                    Dim originalFormat As Object = readProp(textCell, "NumberFormat")
                                    writeProp(textCell, "NumberFormat", "@")
                                    writeProp(textCell, "Value2", expectedValue)
                                    writeProp(textCell, "NumberFormat", originalFormat)
                                    actualValue = readProp(textCell, "Value2")
                                    preservedTextCount += 1
                                Finally
                                    releaseCom(textCell)
                                End Try
                            End If
                            If Not valuesMatch(expectedValue, actualValue) Then
                                Throw New System.InvalidOperationException("Exported value differs from its source at " &
                                    columnName(columnOffset + 1) & (rowOffset + 1).ToString() & " for group " & groupName)
                            End If
                        End If
                    Next
                Next
                If preservedTextCount > 0 Then log("TEXT_VALUES_PRESERVED | Group=" & groupName & " | Cells=" & preservedTextCount.ToString())
            End If
            writeProp(excel, "CutCopyMode", False)

            ' D. Preserve layout without duplicating the row/column copy code.
            currentStep = "Preserve column widths and row heights"
            copyLayout(sourceSheet, reportSheet, "Columns", "ColumnWidth", lastColumn)
            copyLayout(sourceSheet, reportSheet, "Rows", "RowHeight", lastRow)

            ' E. Repair an empty header and distinguish missing sheets from empty data.
            If Not missingSource AndAlso repairEmptyHeaderRow AndAlso sourceHeaders(groupName)(0) = 0 Then
                currentStep = "Restore the empty header row"
                Dim templateSheet As Object = Nothing, templateRange As Object = Nothing, destinationHeader As Object = Nothing
                Try
                    templateSheet = itemAt(sourceSheets, headerTemplateName)
                    Dim headerAddress As String = "A" & headerRow.ToString() & ":" & columnName(headerTemplateColumnCount) & headerRow.ToString()
                    templateRange = rangeAt(templateSheet, headerAddress)
                    destinationHeader = rangeAt(reportSheet, headerAddress)
                    xlMethod(templateRange, "Copy", New Object() {destinationHeader}, New String() {"Destination"})
                    If in_ValuesOnly Then
                        writeProp(destinationHeader, "Value2", readProp(templateRange, "Value2"))
                    End If
                    repairedHeaderCount += 1
                    log("HEADER_ROW_REPAIRED | Group=" & groupName & " | HeaderRow=" & headerRow.ToString() & " | Template=" & headerTemplateName)
                Finally
                    releaseAll(New Object() {destinationHeader, templateRange, templateSheet})
                End Try
            End If
            If missingSource Then
                Dim warningCell As Object = Nothing
                Try
                    warningCell = rangeAt(reportSheet, "A1")
                    Dim originalTitle As String = System.Convert.ToString(readProp(warningCell, "Value2"))
                    writeProp(warningCell, "Value2", originalTitle & " | SOURCE SHEET MISSING: " & groupName)
                Finally
                    releaseCom(warningCell)
                End Try
            ElseIf sourceDataRows = 0 Then
                emptyGroupCount += 1
                log("EMPTY_SOURCE_DATA | Group=" & groupName & " | Action=HeadersOnly")
            End If

            ' F. Delete the old report only after its replacement is complete and visible.
            ' Retain every other report and order Report 1, 2, 3, 4 even on recovery runs.
            currentStep = "Replace and name the current report sheet"
            If previousNames.Contains(reportName) Then
                oldSheet = itemAt(destinationSheets, reportName)
                deleteSheetSafely(destinationBook, oldSheet, reportSheet)
            End If
            writeProp(reportSheet, "Name", reportName)
            currentStep = "Order the report sheets"
            Dim namesNow As System.Collections.Generic.HashSet(Of String) = sheetNames(destinationBook)
            For number As Integer = 4 To 1 Step -1
                Dim name As String = "Report " & number.ToString()
                If Not namesNow.Contains(name) Then Continue For
                Dim firstSheet As Object = Nothing, movingSheet As Object = Nothing
                Try
                    firstSheet = itemAt(destinationSheets, 1)
                    If String.Equals(System.Convert.ToString(readProp(firstSheet, "Name")), name, ignoreCase) Then Continue For
                    movingSheet = itemAt(destinationSheets, name)
                    xlMethod(movingSheet, "Move", New Object() {firstSheet}, New String() {"Before"})
                Finally
                    releaseAll(New Object() {movingSheet, firstSheet})
                End Try
            Next
            Dim resultNames As System.Collections.Generic.HashSet(Of String) = sheetNames(destinationBook)
            If Not resultNames.Contains(reportName) Then Throw New System.InvalidOperationException("The report worksheet is missing before save.")
            Dim expectedCount As Integer = previousNames.Count + If(previousNames.Contains(reportName), 0, 1)
            If resultNames.Count <> expectedCount Then Throw New System.InvalidOperationException("Unexpected final worksheet count: " & fileName)
            For Each previousName As String In previousNames
                If Not resultNames.Contains(previousName) Then Throw New System.InvalidOperationException("An existing worksheet would be lost: " & previousName)
            Next
            log("TRANSFER_VERIFIED | Group=" & groupName & " | Sheet=" & reportName & " | DataRows=" & sourceDataRows.ToString() &
                " | MissingSource=" & missingSource.ToString() & " | FinalSheetCount=" & resultNames.Count.ToString())
            currentStep = "Save the staged group workbook"
            If existed Then
                runMethod(destinationBook, "Save")
            Else
                xl(destinationBook, "SaveAs", Microsoft.VisualBasic.CallType.Method, New Object() {stagedPath, 51})
            End If
            closeBook(destinationBook)
            releaseCom(destinationBook)
            destinationBook = Nothing
            If Not System.IO.File.Exists(stagedPath) OrElse (New System.IO.FileInfo(stagedPath)).Length = 0 Then
                Throw New System.IO.IOException("Excel did not create the staged workbook: " & fileName)
            End If
            log("STAGED | Group=" & groupName & " | Sheet=" & reportName)
        Finally
            releaseAll(New Object() {oldSheet, reportSheet, anchorSheet, destinationRange, sourceRange, sourceAreaColumns, sourceAreaRows, sourceArea, sourceSheet, sourceSheets, destinationSheets})
        End Try
    Next

    ' 10. CLOSE OUR EXCEL INSTANCE BEFORE PUBLISHING FILES
    currentStep = "Close export Excel instance"
    closeBook(sourceBook)
    releaseCom(sourceBook)
    sourceBook = Nothing
    closeBook(groupsBook)
    releaseCom(groupsBook)
    groupsBook = Nothing
    runMethod(excel, "Quit")
    releaseCom(workbooks)
    workbooks = Nothing
    releaseCom(excel)
    excel = Nothing

    ' 11. PUBLISH VERIFIED FILES WITH BACKUPS
    ' Publication is per file, not one transaction across every group.
    ' If a later file fails, rerunning safely replaces this report's sheet again.
    currentStep = "Publish group workbooks"
    For Each groupName As String In groups
        currentGroup = groupName
        Dim fileName As String = "New Report - " & groupName & ".xlsx"
        Dim finalPath As String = System.IO.Path.Combine(periodFolder, fileName)
        Dim stagedPath As String = System.IO.Path.Combine(stageFolder, fileName)
        Dim originalHash As String = expectedHashes(finalPath)
        If originalHash Is Nothing Then
            ' Move does not overwrite a file that unexpectedly appeared.
            System.IO.File.Move(stagedPath, finalPath)
        Else
            If Not System.IO.File.Exists(finalPath) Then Throw New System.IO.IOException("Output workbook disappeared during export: " & fileName)
            Using guard As New System.IO.FileStream(finalPath, System.IO.FileMode.Open, System.IO.FileAccess.ReadWrite, System.IO.FileShare.None)
                If Not String.Equals(hashStream(guard), originalHash, System.StringComparison.Ordinal) Then
                    Throw New System.IO.IOException("Output workbook changed during export. Rerun after resolving the concurrent edit: " & fileName)
                End If
            End Using
            System.IO.Directory.CreateDirectory(backupFolder)
            ' No Delete + overwrite fallback. If Replace is unsupported on the share, fail safely.
            System.IO.File.Replace(stagedPath, finalPath, System.IO.Path.Combine(backupFolder, fileName), False)
        End If
        completedFiles.Add(finalPath)
        log("COMMITTED | Group=" & groupName & " | Sheet=" & reportName)
    Next

    out_UpdatedFiles = completedFiles.ToArray()
    out_Summary = "RunId=" & runId & " | Period=" & reportMonth & " | InputPeriod=" & periodInput & " | " & reportName &
                  " | UpdatedFiles=" & completedFiles.Count.ToString() &
                  " | MissingSheets=" & missingSheetCount.ToString() & " | RepairedHeaders=" & repairedHeaderCount.ToString() &
                  " | EmptyGroups=" & emptyGroupCount.ToString() & " | DuplicateNamesIgnored=" & duplicateCount.ToString()
    log("SUCCESS | UpdatedFiles=" & completedFiles.Count.ToString() & " | MissingSheets=" & missingSheetCount.ToString() &
        " | RepairedHeaders=" & repairedHeaderCount.ToString() & " | EmptyGroups=" & emptyGroupCount.ToString())
    success = True
' 12. FAILURE — include the exact group/step and preserve the original exception.
Catch ex As System.Exception
    Try
        If System.IO.Directory.Exists(journalFolder) Then
            log("FAILED | Group=" & currentGroup & " | Step=" & currentStep & " | Type=" & ex.GetType().FullName &
                " | Message=" & ex.Message.Replace(System.Environment.NewLine, " ") &
                " | CommittedFiles=" & completedFiles.Count.ToString())
        End If
    Catch
        ' Do not hide the original failure if writing the failure log also fails.
    End Try
    Throw
Finally
    ' 13. CLEANUP — touch only objects created here; never kill the user's Excel.
    ' Suppress cleanup errors only. Failed staging remains available for investigation.
    For Each book As Object In New Object() {destinationBook, sourceBook, groupsBook}
        If book Is Nothing Then Continue For
        Try
            closeBook(book)
        Catch
        Finally
            Try
                releaseCom(book)
            Catch
            End Try
        End Try
    Next
    If excel IsNot Nothing Then
        Try
            runMethod(excel, "Quit")
        Catch
        End Try
    End If
    Try
        releaseCom(workbooks)
    Catch
    End Try
    Try
        releaseCom(excel)
    Catch
    End Try
    If success Then
        Try
            System.IO.Directory.Delete(stageFolder, True)
        Catch
            ' Final files are already published; leftover staging can be removed later.
        End Try
    End If
    If lockStream IsNot Nothing Then lockStream.Dispose()
End Try
