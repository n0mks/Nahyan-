  function sanitizeRelativePath(path){







    var decodedPath =



      decodeAndRepair(



        path



      )







      .replace(



        /\\/g,



        '/'



      )







      .replace(



        /^\/+/,



        ''



      )







      .replace(



        /\/{2,}/g,



        '/'



      );











    var source =



      decodedPath.split('/');











    var result =



      [];











    for(



      var i = 0;



      i < source.length;



      i++



    ){







      var segment =



        String(



          source[i] ||



          ''



        ).trim();











      if(



        !segment ||



        segment === '.'



      ){







        continue;







      }











      if(



        segment === '..'



      ){







        if(



          result.length



        ){







          result.pop();







        }







        continue;







      }











      result.push(



        segment



      );







    }











    return result.join('/');







  }

  function extractCircularRelativePath(rawValue){

    }











    var decoded =



      decodeAndRepair(

      )







      .replace(



        /\\/g,



        '/'



      )







      .trim();











    if(



      !decoded



    ){







      return '';







    }











    /*



      Existing SharePoint URL



    */







    if(



      SHAREPOINT_BASE_DECODED



    ){







      var lowerDecoded =



        decoded.toLowerCase();











      var lowerBase =



        SHAREPOINT_BASE_DECODED.toLowerCase();











      if(



        lowerDecoded.indexOf(



          lowerBase

      ){







        return sanitizeRelativePath(







          decoded.substring(



            SHAREPOINT_BASE_DECODED.length



          )







        );







      }







    }











    /*



      Remove file:// / file:/// prefixes.



    */







    decoded =



      decoded.replace(



        /^file:\/*/i,



        ''



      );











    /*



      Normalize all slashes.



    */







    decoded =



      decoded.replace(



        /\/{2,}/g,



        '/'



      );











    decoded =



      decoded.replace(



        /^\/+/,



        ''



      );











    var lower =



      decoded.toLowerCase();











    /*



      All known historical source structures.



    */







    var markers = [







      'oabstg/compliance/circulars/cbo/cbocirculars/',







      'oabstg2/compliance/circulars/cbo/cbocirculars/',







      'compliance/circulars/cbo/cbocirculars/',







      'oabstg/compliance/circulars/cbo/cbo circulars/',







      'oabstg2/compliance/circulars/cbo/cbo circulars/',







      'compliance/circulars/cbo/cbo circulars/',







      'oabstg/cbo/cbo circulars/',







      'oabstg2/cbo/cbo circulars/',







      'cbo/cbocirculars/',







      'cbo/cbo circulars/',







      'cbocirculars/',







      'cbo circulars/'







    ];











    for(



      var i = 0;



      i < markers.length;



      i++



    ){







      var index =



        lower.indexOf(



          markers[i]



        );











      if(



        index >= 0



      ){







        return sanitizeRelativePath(







          decoded.substring(







            index +



            markers[i].length







          )







        );







      }







    }











    /*



      Flexible CBO/CBOCirculars matcher.

    var flexible =



      decoded.match(







        /(?:^|\/)cbo\/cbo\s*circulars\/(.+)$/i







      );











    if(



      flexible &&



      flexible[1]

      return sanitizeRelativePath(



        flexible[1]



      );







    }











    /*



      Flexible compliance path.



    */







    var complianceMatch =



      decoded.match(







        /(?:^|\/)compliance\/circulars\/cbo\/cbo\s*circulars\/(.+)$/i

      decoded.indexOf('/') < 0

        decoded



      );

  /* =======================================================



     DATE NORMALIZATION







     10-Aug-26 -> 10-08-2026



     6-Aug-26  -> 06-08-2026



  ======================================================== */







  function normalizeDateForFile(value){







    var text =



      decodeAndRepair(



        value



      );











    if(



      !text



    ){







      return '';







    }











    var months = {







      jan:'01',



      feb:'02',



      mar:'03',



      apr:'04',



      may:'05',



      jun:'06',



      jul:'07',



      aug:'08',



      sep:'09',



      oct:'10',



      nov:'11',



      dec:'12'







    };











    var match =



      text.match(







        /^(\d{1,2})[-\/.\s]+([A-Za-z]{3,9})[-\/.\s]+(\d{2,4})$/







      );











    if(



      match



    ){







      var monthName =



        match[2]



          .substring(0,3)



          .toLowerCase();











      if(



        months[monthName]



      ){







        var year =



          match[3];











        if(



          year.length === 2



        ){







          year =



            '20' + year;







        }











        return (



          ('0' + match[1]).slice(-2)



          +



          '-'



          +



          months[monthName]



          +



          '-'



          +



          year



        );







      }







    }











    match =



      text.match(







        /^(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{2,4})$/







      );











    if(



      match



    ){







      var numericYear =



        match[3];











      if(



        numericYear.length === 2



      ){







        numericYear =



          '20' +



          numericYear;







      }











      return (



        ('0' + match[1]).slice(-2)



        +



        '-'



        +



        ('0' + match[2]).slice(-2)



        +



        '-'



        +



        numericYear



      );







    }











    return text;







  }

    if(







      typeof cell.w !==



      'undefined'







      &&







      cell.w !== null







      &&







      String(



        cell.w



      ).trim() !== ''







    ){







      return String(



        cell.w



      ).trim();







    }











    if(







      typeof cell.v !==



      'undefined'







      &&







      cell.v !== null







    ){







      return String(



        cell.v



      ).trim();







    }











    return '';







  }











  function extractCellLink(cell){







    if(



      !cell



    ){







      return '';







    }











    /*



      Standard Excel hyperlink



    */







    if(



      cell.l



    ){







      if(



        cell.l.Target



      ){







        return String(



          cell.l.Target



        ).trim();







      }











      if(



        cell.l.target



      ){







        return String(



          cell.l.target



        ).trim();







      }







    }











    /*



      Excel HYPERLINK()



    */







    if(



      cell.f



    ){







      var formula =



        String(



          cell.f



        );











      var match =



        formula.match(







          /^HYPERLINK\s*\(\s*"((?:[^"]|"")+)"/i







        );











      if(



        match &&



        match[1]



      ){







        return match[1]







          .replace(



            /""/g,



            '"'



          )







          .trim();







      }







    }











    var value =



      formatCell(



        cell



      );











    if(



      !value



    ){







      return '';







    }











    /*



      Includes:



      File:///\\oabstg\...



    */







    if(



      /^file:/i.test(



        value



      )



    ){







      return value;







    }











    if(



      /^https?:\/\//i.test(



        value



      )



    ){







      return value;







    }











    if(



      /^\\\\/.test(



        value



      )



    ){







      return value;







    }











    if(



      /^[A-Za-z]:[\\/]/.test(



        value



      )



    ){







      return value;







    }











    var normalized =



      decodeAndRepair(



        value



      )







      .replace(



        /\\/g,



        '/'



      )







      .toLowerCase();











    if(







      normalized.indexOf(



        'cbo/cbo circulars/'



      ) >= 0







      ||







      normalized.indexOf(



        'cbo/cbocirculars/'



      ) >= 0







      ||







      normalized.indexOf(



        'compliance/circulars/cbo/'



      ) >= 0







    ){







      return value;







    }











    return '';







  }











  /* =======================================================



     HEADERS



  ======================================================== */







  function findHeaderRow(



    ws,



    range



  ){







    var best =



      null;











    var maxRow =



      Math.min(



        range.e.r,



        range.s.r + 30



      );











    for(



      var r = range.s.r;



      r <= maxRow;



      r++



    ){







      var map = {







        date:-1,







        circular:-1,







        subject:-1,







        business:-1







      };











      for(



        var c = range.s.c;



        c <= range.e.c;



        c++



      ){







        var cell =



          ws[



            XLSX.utils.encode_cell({



              r:r,



              c:c



            })



          ];











        var key =



          headerKey(



            formatCell(



              cell



            )



          );











        if(



          key === 'date'



        ){







          map.date =



            c;







        }











        if(







          key === 'circularregulation'







          ||







          key === 'circular'







          ||







          key === 'regulation'







          ||







          key.indexOf(



            'circular'



          ) >= 0







          ||







          key.indexOf(



            'regulation'



          ) >= 0







        ){







          map.circular =



            c;







        }











        if(



          key === 'subject'



        ){







          map.subject =



            c;







        }











        if(







          key === 'businessline'







          ||







          key === 'business'







        ){







          map.business =



            c;







        }







      }











      var score =



        0;











      if(



        map.date >= 0



      ){



        score++;



      }











      if(



        map.circular >= 0



      ){



        score++;



      }











      if(



        map.subject >= 0



      ){



        score++;



      }











      if(



        map.business >= 0



      ){



        score++;



      }











      if(



        !best ||



        score > best.score



      ){







        best = {







          row:r,







          score:score,







          map:map







        };







      }











      if(



        score === 4



      ){







        break;







      }







    }











    return (



      best &&



      best.score === 4



    )







      ? best







      : null;







  }

  function buildActualFileUrl(file){

    }











    var serverRelative =



      String(



        file.ServerRelativeUrl ||



        ''



      );











    /*

    if(



      serverRelative &&



      SHAREPOINT_ORIGIN



    ){







      try{







        var decoded =



          safeDecodeRepeated(

          );











        var segments =



          decoded.split('/');











        var encoded =



          segments







          .map(



            function(segment,index){







              if(



                index === 0 &&



                segment === ''



              ){







                return '';







              }











              return encodeURIComponent(



                segment



              );







            }



          )







          .join('/');











        return (



          SHAREPOINT_ORIGIN +



          encoded



        );







      }catch(e){}







    }











    /*



      Fallback using Name.



    */







    if(



      file.Name



    ){







      return buildSharePointUrl(



        file.Name



      );







    }











    return '';







  }

  /* =======================================================



     TOKEN MATCHING



  ======================================================== */







  function tokenize(value){







    var text =



      normalizeFileStem(



        value



      )







      .replace(



        /[^a-z0-9]+/g,



        ' '



      );











    var source =



      text.split(/\s+/);











    var result =



      [];











    var used =



      {};











    for(



      var i = 0;



      i < source.length;



      i++



    ){







      var token =



        source[i];











      if(



        !token ||



        token.length < 2 ||



        used[token]



      ){







        continue;







      }











      used[token] =



        true;











      result.push(



        token



      );







    }











    return result;







  }











  function similarity(a,b){







    var aTokens =



      tokenize(



        a



      );











    var bTokens =



      tokenize(



        b



      );











    if(



      !aTokens.length ||



      !bTokens.length



    ){







      return 0;







    }











    var map =



      {};











    for(



      var i = 0;



      i < bTokens.length;



      i++



    ){







      map[



        bTokens[i]



      ] =



        true;







    }











    var common =



      0;











    for(



      var j = 0;



      j < aTokens.length;



      j++



    ){







      if(



        map[



          aTokens[j]



        ]



      ){







        common++;







      }







    }











    return (



      2 *



      common /



      (



        aTokens.length +



        bTokens.length



      )



    );







  }











  /* =======================================================



     SMART ROW RESOLVER







     This is the important new part.



  ======================================================== */







  function resolveRowFile(row){







    if(



      !state.resolverAvailable



    ){







      return null;







    }











    var candidates =



      buildRowFileCandidates(



        row



      );











    /*



      1. Exact compact match.



    */







    for(



      var i = 0;



      i < candidates.length;



      i++



    ){







      var exactKey =



        compactFileKey(



          candidates[i]



        );











      var exactFiles =



        state.compactIndex[



          exactKey



        ];











      if(



        exactFiles &&



        exactFiles.length === 1



      ){







        return exactFiles[0];







      }







    }











    /*



      2. Strong Date + Circular prefix.







      Example:







      Excel:



      10-08-2026SD2026Banks314







      Real file:



      10-08-2026 SD 2026 Banks 314 Treatment...



    */







    var core =



      compactFileKey(







        normalizeDateForFile(



          row.date



        )







        +







        ' '







        +







        String(



          row.circular ||



          ''



        )







      );











    if(



      core.length >= 10



    ){







      var coreMatches =



        [];











      for(



        var c = 0;



        c < state.fileInventory.length;



        c++



      ){







        if(



          state.fileInventory[c]



            .compact



            .indexOf(



              core



            ) === 0



        ){







          coreMatches.push(



            state.fileInventory[c]



          );







        }







      }











      if(



        coreMatches.length === 1



      ){







        return coreMatches[0];







      }







    }











    /*



      3. Prefix matching for every candidate.



    */







    for(



      var p = 0;



      p < candidates.length;



      p++



    ){







      var candidateKey =



        compactFileKey(



          candidates[p]



        );











      if(



        candidateKey.length < 12



      ){







        continue;







      }











      var prefixMatches =



        [];











      for(



        var f = 0;



        f < state.fileInventory.length;



        f++



      ){







        var actualKey =



          state.fileInventory[f]



            .compact;











        if(







          actualKey.indexOf(



            candidateKey



          ) === 0







          ||







          (



            actualKey.length >= 14







            &&







            candidateKey.indexOf(



              actualKey



            ) === 0



          )







        ){







          prefixMatches.push(



            state.fileInventory[f]



          );







        }







      }











      if(



        prefixMatches.length === 1



      ){







        return prefixMatches[0];







      }







    }











    /*



      4. Controlled fuzzy match using the



         reconstructed full filename.



    */







    var fullCandidate =



      [







        normalizeDateForFile(



          row.date



        ),







        row.circular,







        row.subject







      ]







      .filter(



        function(value){



          return !!value;



        }



      )







      .join(' ');











    var best =



      null;











    var bestScore =



      0;











    var secondScore =



      0;











    for(



      var x = 0;



      x < state.fileInventory.length;



      x++



    ){







      var score =



        similarity(







          fullCandidate,







          state.fileInventory[x].Name







        );











      if(



        core &&



        state.fileInventory[x]



          .compact



          .indexOf(



            core



          ) === 0



      ){







        score +=



          0.35;







      }











      if(



        score > bestScore



      ){







        secondScore =



          bestScore;











        bestScore =



          score;











        best =



          state.fileInventory[x];







      }else if(



        score > secondScore



      ){







        secondScore =



          score;







      }







    }











    if(







      best







      &&







      bestScore >= 0.72







      &&







      (



        bestScore -



        secondScore



      ) >= 0.08







    ){







      return best;







    }











    return null;







  }

      }







    }







  }

  /* =======================================================



     BUSINESS FILTER



  ======================================================== */







  function splitBusinessLines(value){







    var text =



      String(



        value ||



        ''



      ).trim();











    if(



      !text



    ){







      return [];







    }











    return text







      .split(



        /[\/;,|]+/



      )







      .map(



        function(value){







          return String(



            value ||



            ''



          ).trim();







        }



      )







      .filter(



        function(value){







          return !!value;







        }



      );







  }











  function rowMatchesBusiness(



    rowBusiness,



    filterValue



  ){







    var target =



      normalizeText(



        filterValue



      );











    if(



      !target



    ){







      return true;







    }











    var items =



      splitBusinessLines(



        rowBusiness



      );











    for(



      var i = 0;



      i < items.length;



      i++



    ){







      if(



        normalizeText(



          items[i]



        ) === target



      ){







        return true;







      }







    }











    return false;







  }











  function populateBusinessFilter(



    rows,



    selected



  ){







    var used =



      {};











    var values =



      [];











    for(



      var i = 0;



      i < rows.length;



      i++



    ){







      var businesses =



        splitBusinessLines(



          rows[i].business



        );











      for(



        var j = 0;



        j < businesses.length;



        j++



      ){







        var key =



          normalizeText(



            businesses[j]



          );











        if(



          !key ||



          used[key]



        ){







          continue;







        }











        used[key] =



          true;











        values.push(



          businesses[j]



        );







      }







    }











    values.sort(



      function(a,b){







        return a.localeCompare(



          b



        );







      }



    );











    businessFilter.innerHTML =







      '<option value="">' +







        'All Business Lines' +







      '</option>';











    for(



      var x = 0;



      x < values.length;



      x++



    ){







      var option =



        document.createElement(



          'option'



        );











      option.value =



        values[x];











      option.textContent =



        values[x];











      businessFilter.appendChild(



        option



      );







    }











    businessFilter.value =



      selected ||



      '';







  }

  /* =======================================================

  function applyFilters(){







    var section =



      getActiveSection();











    section.search =



      searchInput.value ||



      '';











    section.business =



      businessFilter.value ||



      '';











    var query =



      normalizeText(



        section.search



      );











    section.filteredRows =



      section.rows.filter(







        function(row){







          var searchable =



            normalizeText([







              row.date,







              row.circular,







              row.subject,







              row.business







            ].join(' '));











          var matchesSearch =







            !query







            ||







            searchable.indexOf(



              query



            ) >= 0;











          var matchesBusiness =



            rowMatchesBusiness(







              row.business,







              section.business







            );











          return (



            matchesSearch &&



            matchesBusiness



          );







        }







      );











    renderRows(



      section.filteredRows



    );











    updateActiveCounters(







      section.rows.length,







      section.filteredRows.length







    );











    setFooter(







      'Showing ' +







      section.filteredRows.length +







      ' of ' +







      section.rows.length +







      ' records'







    );







  }

  }

  function loadScriptOnce(



    src,



    callback



  ){







    if(



      window.XLSX



    ){







      callback(



        true



      );







      return;







    }











    var scripts =



      document.getElementsByTagName(



        'script'



      );











    for(



      var i = 0;



      i < scripts.length;



      i++



    ){







      if(



        scripts[i]



          .getAttribute(



            'data-rc-src'



          ) === src



      ){







        scripts[i]



          .addEventListener(







            'load',







            function(){







              callback(



                !!window.XLSX



              );







            }







          );











        return;







      }







    }











    var script =



      document.createElement(



        'script'



      );











    script.src =



      src;











    script.async =



      true;











    script.setAttribute(



      'data-rc-src',



      src



    );











    script.onload =



      function(){







        callback(



          !!window.XLSX



        );







      };











    script.onerror =



      function(){







        callback(



          false



        );







      };











    document.head.appendChild(



      script



    );







  }











  /* =======================================================



     FETCH EXCEL



  ======================================================== */







  function fetchArrayBuffer(



    url,



    callback



  ){







    var separator =



      url.indexOf('?') >= 0



        ? '&'



        : '?';











    var requestUrl =







      url +







      separator +







      'v=' +







      new Date().getTime();











    fetch(



      requestUrl,



      {







        method:'GET',







        credentials:'same-origin',







        cache:'no-store'







      }



    )







    .then(



      function(response){







        if(



          !response.ok



        ){







          throw new Error(



            'Excel request failed'



          );







        }











        return response.arrayBuffer();







      }



    )







    .then(



      function(buffer){







        callback(



          buffer



        );







      }



    )







    .catch(



      function(){







        callback(



          null



        );







      }



    );







  }

    }











    if(



      !two.ok

      errors.push(



        two.error

  /* =======================================================

  function loadWorkbook(isRefresh){







    if(



      state.isLoading



    ){







      return;







    }











    if(



      !CONFIG_XLSX ||



      !SHEETJS_SRC ||



      !SHAREPOINT_BASE



    ){







      showMessage(







        'Page configuration is incomplete.',







        'error'







      );











      return;







    }











    setLoadingState(



      true



    );











    setLoading(







      isRefresh







        ? 'Refreshing records...'







        : 'Loading records...'







    );











    setFooter(



      'Loading...'



    );











    showMessage(



      '',



      'info'



    );











    loadScriptOnce(







      SHEETJS_SRC,







      function(loaded){







        if(



          !loaded



        ){







          setLoadingState(



            false



          );











          showMessage(







            'Unable to load the Excel reader library.',







            'error'







          );











          setLoading(



            'Unable to load records.'



          );











          return;







        }











        fetchArrayBuffer(







          CONFIG_XLSX,







          function(buffer){







            if(



              !buffer



            ){







              setLoadingState(



                false



              );











              showMessage(







                'Unable to load the Excel configuration file.',







                'error'







              );











              setLoading(



                'Unable to load records.'



              );











              return;







            }











            try{







              var workbook =



                XLSX.read(







                  new Uint8Array(



                    buffer



                  ),







                  {







                    type:'array',







                    cellFormula:true,







                    cellText:true







                  }







                );











              state.workbook =



                workbook;











              var errors =



                readConfiguredSheets(



                  workbook



                );











              /*



                Read the REAL SharePoint files.



              */







              fetchSharePointFiles(







                function(files){







                  if(



                    files &&



                    files.length



                  ){







                    indexFiles(



                      files



                    );











                    /*



                      Replace old/local/network



                      links with actual SharePoint files.



                    */







                    resolveAllLinks();







                  }











                  renderActiveSection();











                  setLoadingState(



                    false



                  );











                  if(



                    errors.length



                  ){







                    showMessage(







                      errors.join(' '),







                      'error'







                    );







                  }else{







                    showMessage(



                      '',



                      'info'



                    );







                  }











                  var total =







                    state.sections.one.rows.length







                    +







                    state.sections.two.rows.length;











                  setFooter(







                    'Loaded ' +



                    total +



                    ' total records'







                  );







                }







              );











            }catch(e){







              setLoadingState(



                false



              );











              showMessage(







                'The Excel file could not be read.',







                'error'







              );











              setLoading(



                'Unable to read records.'



              );







            }







          }







        );







      }







    );







  }
