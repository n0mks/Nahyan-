' Late-bound COM through CallByName also works with Option Strict On.
' Retry only COM calls Excel explicitly rejected as busy; never retry a whole export here.
Dim xl As System.Func(Of Object, String, Microsoft.VisualBasic.CallType, Object(), Object) =
    Function(target As Object, member As String, callType As Microsoft.VisualBasic.CallType, args As Object()) As Object
        For attempt As Integer = 1 To 3
            Try
                Return Microsoft.VisualBasic.Interaction.CallByName(target, member, callType, args)
            Catch ex As System.Exception
                Dim cause As System.Exception = ex
                While TypeOf cause Is System.Reflection.TargetInvocationException AndAlso cause.InnerException IsNot Nothing
                    cause = cause.InnerException
                End While
                Dim comError As System.Runtime.InteropServices.COMException =
                    TryCast(cause, System.Runtime.InteropServices.COMException)
                If comError Is Nothing OrElse
                   (comError.ErrorCode <> -2147418111 AndAlso comError.ErrorCode <> -2147417846) OrElse
                   attempt = 3 Then
                    Throw
                End If
                System.Threading.Thread.Sleep(250 * attempt)
            End Try
        Next
        Throw New System.InvalidOperationException("Excel COM retry limit reached.")
    End Function

' Invoke Range.Copy and Worksheets.Add/Move with explicit argument names.
Dim xlMethod As System.Func(Of Object, String, Object(), String(), Object) =
    Function(target As Object, member As String, args As Object(), argumentNames As String()) As Object
        For attempt As Integer = 1 To 3
            Try
                Return target.GetType().InvokeMember(member,
                    System.Reflection.BindingFlags.InvokeMethod Or System.Reflection.BindingFlags.Public Or
                    System.Reflection.BindingFlags.Instance Or System.Reflection.BindingFlags.OptionalParamBinding,
                    Nothing, target, args, Nothing,
                    System.Globalization.CultureInfo.GetCultureInfo("en-US"), argumentNames)
            Catch ex As System.Exception
                Dim cause As System.Exception = ex
                While TypeOf cause Is System.Reflection.TargetInvocationException AndAlso cause.InnerException IsNot Nothing
                    cause = cause.InnerException
                End While
                Dim comError As System.Runtime.InteropServices.COMException =
                    TryCast(cause, System.Runtime.InteropServices.COMException)
                If comError Is Nothing OrElse
                   (comError.ErrorCode <> -2147418111 AndAlso comError.ErrorCode <> -2147417846) OrElse
                   attempt = 3 Then
                    Throw
                End If
                System.Threading.Thread.Sleep(250 * attempt)
            End Try
        Next
        Throw New System.InvalidOperationException("Excel COM method retry limit reached.")
    End Function

Dim releaseCom As System.Action(Of Object) =
    Sub(obj As Object)
        If obj IsNot Nothing AndAlso System.Runtime.InteropServices.Marshal.IsComObject(obj) Then
            ' Balance each COM reference obtained above. Do not FinalRelease shared RCWs:
            ' Excel can return the same RCW for several references to one sheet/collection.
            System.Runtime.InteropServices.Marshal.ReleaseComObject(obj)
        End If
    End Sub

Dim openBook As System.Func(Of Object, String, Boolean, Object) =
    Function(books As Object, path As String, openReadOnly As Boolean) As Object
        ' Open without updating external links, with macros disabled at Application level.
        Return xl(books, "Open", Microsoft.VisualBasic.CallType.Method,
            New Object() {path, 0, openReadOnly, System.Type.Missing, "", "", True,
                          System.Type.Missing, System.Type.Missing, False, False,
                          System.Type.Missing, False, True, 0})
    End Function

Dim sheetNames As System.Func(Of Object, System.Collections.Generic.HashSet(Of String)) =
    Function(book As Object) As System.Collections.Generic.HashSet(Of String)
        Dim names As New System.Collections.Generic.HashSet(Of String)(System.StringComparer.OrdinalIgnoreCase)
        Dim sheets As Object = Nothing
        Try
            sheets = xl(book, "Worksheets", Microsoft.VisualBasic.CallType.Get, New Object() {})
            Dim count As Integer = System.Convert.ToInt32(xl(sheets, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            For index As Integer = 1 To count
                Dim sheet As Object = Nothing
                Try
                    sheet = xl(sheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {index})
                    names.Add(System.Convert.ToString(xl(sheet, "Name", Microsoft.VisualBasic.CallType.Get, New Object() {})))
                Finally
                    releaseCom(sheet)
                End Try
            Next
        Finally
            releaseCom(sheets)
        End Try
        Return names
    End Function

Dim columnName As System.Func(Of Integer, String) =
    Function(columnNumber As Integer) As String
        Dim name As String = String.Empty
        Do While columnNumber > 0
            columnNumber -= 1
            name = System.Convert.ToChar(65 + (columnNumber Mod 26)).ToString() & name
            columnNumber = columnNumber \ 26
        Loop
        Return name
    End Function

Dim valueAt As System.Func(Of Object, Integer, Integer, Object) =
    Function(raw As Object, rowOffset As Integer, columnOffset As Integer) As Object
        Dim matrix As System.Array = TryCast(raw, System.Array)
        If matrix IsNot Nothing Then
            If matrix.Rank <> 2 Then Throw New System.InvalidOperationException("Excel returned an unexpected value array.")
            Return matrix.GetValue(matrix.GetLowerBound(0) + rowOffset, matrix.GetLowerBound(1) + columnOffset)
        End If
        If rowOffset = 0 AndAlso columnOffset = 0 Then Return raw
        Return Nothing
    End Function

Dim hasValue As System.Func(Of Object, Boolean) =
    Function(value As Object) As Boolean
        If value Is Nothing OrElse value Is System.DBNull.Value Then Return False
        If TypeOf value Is String Then Return DirectCast(value, String).Length > 0
        Return True
    End Function

Dim valuesMatch As System.Func(Of Object, Object, Boolean) =
    Function(left As Object, right As Object) As Boolean
        If Not hasValue(left) AndAlso Not hasValue(right) Then Return True
        Dim leftError As System.Runtime.InteropServices.ErrorWrapper = TryCast(left, System.Runtime.InteropServices.ErrorWrapper)
        Dim rightError As System.Runtime.InteropServices.ErrorWrapper = TryCast(right, System.Runtime.InteropServices.ErrorWrapper)
        If leftError IsNot Nothing OrElse rightError IsNot Nothing Then
            Return leftError IsNot Nothing AndAlso rightError IsNot Nothing AndAlso leftError.ErrorCode = rightError.ErrorCode
        End If
        Return System.Object.Equals(left, right)
    End Function

Dim headerInfo As System.Func(Of Object, Integer()) =
    Function(sheet As Object) As Integer()
        Dim area As Object = Nothing
        Dim areaColumns As Object = Nothing
        Dim headers As Object = Nothing
        Try
            area = xl(sheet, "UsedRange", Microsoft.VisualBasic.CallType.Get, New Object() {})
            areaColumns = xl(area, "Columns", Microsoft.VisualBasic.CallType.Get, New Object() {})
            Dim lastColumn As Integer = System.Convert.ToInt32(xl(area, "Column", Microsoft.VisualBasic.CallType.Get, New Object() {})) +
                System.Convert.ToInt32(xl(areaColumns, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {})) - 1
            headers = xl(sheet, "Range", Microsoft.VisualBasic.CallType.Get,
                New Object() {"A" & headerRow.ToString() & ":" & columnName(lastColumn) & headerRow.ToString()})
            Dim raw As Object = xl(headers, "Value2", Microsoft.VisualBasic.CallType.Get, New Object() {})
            Dim labelCount As Integer = 0
            For columnOffset As Integer = 0 To lastColumn - 1
                Dim label As Object = valueAt(raw, 0, columnOffset)
                If TypeOf label Is String AndAlso Not String.IsNullOrWhiteSpace(DirectCast(label, String)) Then labelCount += 1
            Next
            Return New Integer() {labelCount, lastColumn}
        Finally
            releaseCom(headers)
            releaseCom(areaColumns)
            releaseCom(area)
        End Try
    End Function

Dim hashStream As System.Func(Of System.IO.Stream, String) =
    Function(stream As System.IO.Stream) As String
        stream.Position = 0
        Using sha As System.Security.Cryptography.SHA256 = System.Security.Cryptography.SHA256.Create()
            Return System.Convert.ToBase64String(sha.ComputeHash(stream))
        End Using
    End Function

' Allow other readers, but reject a file held for writing by Excel or another process.
' This checks readiness; it cannot close a WorkbookApplication owned by the caller.
Dim waitForInput As System.Action(Of String) =
    Sub(lockedInputPath As String)
        Dim timer As System.Diagnostics.Stopwatch = System.Diagnostics.Stopwatch.StartNew()
        Dim loggedWait As Boolean = False
        Do
            Try
                Using probe As New System.IO.FileStream(lockedInputPath, System.IO.FileMode.Open,
                            System.IO.FileAccess.Read, System.IO.FileShare.Read)
                    If probe.Length = 0 Then
                        Throw New System.IO.InvalidDataException("Input workbook is empty: " & System.IO.Path.GetFileName(lockedInputPath))
                    End If
                End Using
                If loggedWait Then log("INPUT_READY | File=" & System.IO.Path.GetFileName(lockedInputPath))
                Exit Do
            Catch ex As System.IO.IOException
                Dim nativeError As Integer = ex.HResult And &HFFFF
                If nativeError <> 32 AndAlso nativeError <> 33 Then Throw
                Dim remainingMilliseconds As Double =
                    CDbl(inputWaitTimeoutSeconds) * 1000.0 - CDbl(timer.ElapsedMilliseconds)
                If remainingMilliseconds <= 0 Then
                    ' No InnerException: the caller's root-exception loop must retain this instruction.
                    Throw New System.IO.IOException(
                        "Input workbook is still locked after " & inputWaitTimeoutSeconds.ToString() &
                        " seconds: " & lockedInputPath &
                        ". Save and close the workbook in the old process. Run aggregation outside every Excel Application Scope that uses this file.")
                End If
                If Not loggedWait Then
                    log("WAITING_FOR_INPUT | File=" & System.IO.Path.GetFileName(lockedInputPath) &
                        " | TimeoutSeconds=" & inputWaitTimeoutSeconds.ToString())
                    loggedWait = True
                End If
                System.Threading.Thread.Sleep(CInt(System.Math.Min(CDbl(inputPollMilliseconds),
                    System.Math.Ceiling(remainingMilliseconds))))
            End Try
        Loop
    End Sub

' The keeper must belong to this workbook and stay visible before deleting any sheet.
Dim deleteSheetSafely As System.Action(Of Object, Object, Object) =
    Sub(book As Object, sheetToDelete As Object, keeper As Object)
        Dim keeperParent As Object = Nothing
        Dim deleteParent As Object = Nothing
        Try
            Dim bookName As String = System.Convert.ToString(xl(book, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            keeperParent = xl(keeper, "Parent", Microsoft.VisualBasic.CallType.Get, New Object() {})
            deleteParent = xl(sheetToDelete, "Parent", Microsoft.VisualBasic.CallType.Get, New Object() {})
            If Not String.Equals(bookName,
                    System.Convert.ToString(xl(keeperParent, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {})),
                    System.StringComparison.OrdinalIgnoreCase) OrElse
               Not String.Equals(bookName,
                    System.Convert.ToString(xl(deleteParent, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {})),
                    System.StringComparison.OrdinalIgnoreCase) Then
                Throw New System.InvalidOperationException("Sheet deletion references the wrong workbook.")
            End If
            Dim keeperName As String = System.Convert.ToString(xl(keeper, "Name", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            Dim deleteName As String = System.Convert.ToString(xl(sheetToDelete, "Name", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            If String.Equals(keeperName, deleteName, System.StringComparison.OrdinalIgnoreCase) Then
                Throw New System.InvalidOperationException("The copied report sheet must not be deleted.")
            End If
            xl(keeper, "Visible", Microsoft.VisualBasic.CallType.Let, New Object() {-1})
            If System.Convert.ToInt32(xl(keeper, "Visible", Microsoft.VisualBasic.CallType.Get, New Object() {})) <> -1 Then
                Throw New System.InvalidOperationException("The copied report sheet is not visible before deletion.")
            End If
            xl(sheetToDelete, "Delete", Microsoft.VisualBasic.CallType.Method, New Object() {})
        Finally
            releaseCom(deleteParent)
            releaseCom(keeperParent)
        End Try
    End Sub

            ". Supply at least one group sheet with this report's real column headers.")
    End If
    log("VALIDATED | Groups=" & groups.Count.ToString() & " | MissingSheets=" & missingGroups.Count.ToString() &
        " | HeaderTemplate=" & headerTemplateName & " | DuplicateNamesIgnored=" & duplicateCount.ToString())
    If ignoredSheets.Count > 0 Then log("IGNORED_SOURCE_SHEETS | Names=" & String.Join(", ", ignoredSheets.ToArray()))

    ' All group files are staged before any final file is updated.
    For Each groupName As String In groups
        currentGroup = groupName
        currentStep = "Prepare destination workbook"
        Dim fileName As String = "New Report - " & groupName & ".xlsx"
        Dim finalPath As String = System.IO.Path.Combine(periodFolder, fileName)
        Dim stagedPath As String = System.IO.Path.Combine(stageFolder, fileName)
        Dim existed As Boolean = System.IO.File.Exists(finalPath)
        Dim missingSource As Boolean = Not sourceNames.Contains(groupName)
        Dim sourceSheetName As String = If(missingSource, headerTemplateName, groupName)
        Dim previousNames As New System.Collections.Generic.HashSet(Of String)(System.StringComparer.OrdinalIgnoreCase)
        Dim destinationSheets As Object = Nothing
        Dim sourceSheets As Object = Nothing
        Dim sourceSheet As Object = Nothing
        Dim sourceArea As Object = Nothing
        Dim sourceAreaRows As Object = Nothing
        Dim sourceAreaColumns As Object = Nothing
        Dim sourceRange As Object = Nothing
        Dim destinationRange As Object = Nothing
        Dim destinationRows As Object = Nothing
        Dim sourceRows As Object = Nothing
        Dim destinationColumns As Object = Nothing
        Dim sourceColumns As Object = Nothing
        Dim anchorSheet As Object = Nothing
        Dim reportSheet As Object = Nothing
        Dim oldSheet As Object = Nothing
        Dim sourceDataRows As Integer = 0
        Try
            If existed Then
                If (System.IO.File.GetAttributes(finalPath) And System.IO.FileAttributes.ReadOnly) <> 0 Then
                    Throw New System.UnauthorizedAccessException("Output workbook is read-only: " & fileName)
                End If
                Using existing As New System.IO.FileStream(finalPath, System.IO.FileMode.Open,
                            System.IO.FileAccess.Read, System.IO.FileShare.None)
                    expectedHashes.Add(finalPath, hashStream(existing))
                    existing.Position = 0
                    Using staged As New System.IO.FileStream(stagedPath, System.IO.FileMode.CreateNew,
                                System.IO.FileAccess.Write, System.IO.FileShare.None)
                        existing.CopyTo(staged)
                        staged.Flush(True)
                    End Using
                End Using
                destinationBook = openBook(workbooks, stagedPath, False)
                previousNames = sheetNames(destinationBook)
                If System.Convert.ToBoolean(xl(destinationBook, "ProtectStructure", Microsoft.VisualBasic.CallType.Get, New Object() {})) Then
                    Throw New System.IO.InvalidDataException("Output workbook structure is protected: " & fileName)
                End If
                If System.Convert.ToBoolean(xl(destinationBook, "Date1904", Microsoft.VisualBasic.CallType.Get, New Object() {})) <> sourceDate1904 Then
                    Throw New System.IO.InvalidDataException("Source and output workbook date systems differ: " & fileName)
                End If
            Else
                expectedHashes.Add(finalPath, Nothing)
                destinationBook = xl(workbooks, "Add", Microsoft.VisualBasic.CallType.Method, New Object() {-4167})
                xl(destinationBook, "Date1904", Microsoft.VisualBasic.CallType.Let, New Object() {sourceDate1904})
            End If
            If System.Convert.ToBoolean(xl(destinationBook, "ReadOnly", Microsoft.VisualBasic.CallType.Get, New Object() {})) Then
                Throw New System.UnauthorizedAccessException("Staged output is read-only: " & fileName)
            End If

            currentStep = "Create an explicit report worksheet"
            destinationSheets = xl(destinationBook, "Worksheets", Microsoft.VisualBasic.CallType.Get, New Object() {})
            Dim beforeCount As Integer = System.Convert.ToInt32(xl(destinationSheets, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            If existed Then
                xl(destinationBook, "Activate", Microsoft.VisualBasic.CallType.Method, New Object() {})
                For index As Integer = 1 To beforeCount
                    Dim candidate As Object = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {index})
                    Try
                        If System.Convert.ToInt32(xl(candidate, "Visible", Microsoft.VisualBasic.CallType.Get, New Object() {})) = -1 Then
                            anchorSheet = candidate
                            candidate = Nothing
                            Exit For
                        End If
                    Finally
                        releaseCom(candidate)
                    End Try
                Next
                If anchorSheet Is Nothing Then Throw New System.IO.InvalidDataException("Output has no visible worksheet: " & fileName)
                reportSheet = xlMethod(destinationSheets, "Add", New Object() {anchorSheet, 1, -4167},
                    New String() {"Before", "Count", "Type"})
            Else
                If beforeCount <> 1 Then Throw New System.InvalidOperationException("New workbook must contain exactly one worksheet.")
                reportSheet = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {1})
            End If
            If reportSheet Is Nothing Then Throw New System.InvalidOperationException("Excel did not return the report worksheet.")
            xl(reportSheet, "Visible", Microsoft.VisualBasic.CallType.Let, New Object() {-1})
            Dim afterCount As Integer = System.Convert.ToInt32(xl(destinationSheets, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            If afterCount <> beforeCount + If(existed, 1, 0) Then Throw New System.InvalidOperationException("Unexpected worksheet count after creating the report sheet.")
            Dim parent As Object = Nothing
            Try
                parent = xl(reportSheet, "Parent", Microsoft.VisualBasic.CallType.Get, New Object() {})
                If Not String.Equals(System.Convert.ToString(xl(parent, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {})),
                    System.Convert.ToString(xl(destinationBook, "FullName", Microsoft.VisualBasic.CallType.Get, New Object() {})),
                    System.StringComparison.OrdinalIgnoreCase) Then
                    Throw New System.InvalidOperationException("The report worksheet belongs to the wrong workbook.")
                End If
            Finally
                releaseCom(parent)
            End Try

            currentStep = "Read the source group cells"
            sourceSheets = xl(sourceBook, "Worksheets", Microsoft.VisualBasic.CallType.Get, New Object() {})
            sourceSheet = xl(sourceSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {sourceSheetName})
            If Not missingSource AndAlso System.Convert.ToBoolean(xl(sourceSheet, "FilterMode", Microsoft.VisualBasic.CallType.Get, New Object() {})) Then
                Throw New System.IO.InvalidDataException("The group sheet still has an active filter. Finalize this group's data before export: " & groupName)
            End If
            sourceArea = xl(sourceSheet, "UsedRange", Microsoft.VisualBasic.CallType.Get, New Object() {})
            sourceAreaRows = xl(sourceArea, "Rows", Microsoft.VisualBasic.CallType.Get, New Object() {})
            sourceAreaColumns = xl(sourceArea, "Columns", Microsoft.VisualBasic.CallType.Get, New Object() {})
            Dim areaFirstRow As Integer = System.Convert.ToInt32(xl(sourceArea, "Row", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            Dim areaFirstColumn As Integer = System.Convert.ToInt32(xl(sourceArea, "Column", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            Dim areaRowCount As Integer = System.Convert.ToInt32(xl(sourceAreaRows, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            Dim areaColumnCount As Integer = System.Convert.ToInt32(xl(sourceAreaColumns, "Count", Microsoft.VisualBasic.CallType.Get, New Object() {}))
            Dim lastColumn As Integer = areaFirstColumn + areaColumnCount - 1
            Dim lastRow As Integer = headerRow
            If missingSource Then
                lastColumn = headerTemplateColumnCount
                missingSheetCount += 1
                log("MISSING_SOURCE_SHEET | Group=" & groupName & " | Action=HeadersOnly | HeaderTemplate=" & headerTemplateName)
            Else
                Dim rawContent As Object = xl(sourceArea, If(in_ValuesOnly, "Value2", "Formula"), Microsoft.VisualBasic.CallType.Get, New Object() {})
                For rowOffset As Integer = 0 To areaRowCount - 1
                    Dim rowHasContent As Boolean = False
                    For columnOffset As Integer = 0 To areaColumnCount - 1
                        If hasValue(valueAt(rawContent, rowOffset, columnOffset)) Then
                            rowHasContent = True
                            Exit For
                        End If
                    Next
                    If rowHasContent Then lastRow = System.Math.Max(lastRow, areaFirstRow + rowOffset)
                Next
            End If
            ' Read from A1 to keep the original title and header row positions.
            Dim address As String = "A1:" & columnName(lastColumn) & lastRow.ToString()
            sourceRange = xl(sourceSheet, "Range", Microsoft.VisualBasic.CallType.Get, New Object() {address})
            destinationRange = xl(reportSheet, "Range", Microsoft.VisualBasic.CallType.Get, New Object() {address})
            Dim sourceValues As Object = xl(sourceRange, "Value2", Microsoft.VisualBasic.CallType.Get, New Object() {})
            For rowOffset As Integer = headerRow To lastRow - 1
                Dim populated As Boolean = False
                For columnOffset As Integer = 0 To lastColumn - 1
                    If hasValue(valueAt(sourceValues, rowOffset, columnOffset)) Then
                        populated = True
                        Exit For
                    End If
                Next
                If populated Then sourceDataRows += 1
            Next

            currentStep = "Transfer cells to the explicit destination range"
            log("TRANSFER_BEGIN | Group=" & groupName & " | SourceSheet=" & sourceSheetName &
                " | Range=" & address & " | DataRows=" & sourceDataRows.ToString() & " | ExistingFile=" & existed.ToString())
            xlMethod(sourceRange, "Copy", New Object() {destinationRange}, New String() {"Destination"})
            If in_ValuesOnly Then
                ' Write the source's saved values directly: no PasteSpecial or recalculation.
                xl(destinationRange, "Value2", Microsoft.VisualBasic.CallType.Let, New Object() {sourceValues})
                currentStep = "Verify every exported cell value"
                Dim actualValues As Object = xl(destinationRange, "Value2", Microsoft.VisualBasic.CallType.Get, New Object() {})
                Dim preservedTextCount As Integer = 0
                For rowOffset As Integer = 0 To lastRow - 1
                    For columnOffset As Integer = 0 To lastColumn - 1
                        Dim expectedValue As Object = valueAt(sourceValues, rowOffset, columnOffset)
                        Dim actualValue As Object = valueAt(actualValues, rowOffset, columnOffset)
                        If Not valuesMatch(expectedValue, actualValue) Then
                            ' Excel can coerce numeric-looking text under General formatting.
                            ' Keep leading zeroes and literal text, then restore its original format.
                            If TypeOf expectedValue Is String Then
                                Dim textCell As Object = Nothing
                                Try
                                    textCell = xl(reportSheet, "Range", Microsoft.VisualBasic.CallType.Get,
                                        New Object() {columnName(columnOffset + 1) & (rowOffset + 1).ToString()})
                                    Dim originalFormat As Object = xl(textCell, "NumberFormat", Microsoft.VisualBasic.CallType.Get, New Object() {})
                                    xl(textCell, "NumberFormat", Microsoft.VisualBasic.CallType.Let, New Object() {"@"})
                                    xl(textCell, "Value2", Microsoft.VisualBasic.CallType.Let, New Object() {expectedValue})
                                    xl(textCell, "NumberFormat", Microsoft.VisualBasic.CallType.Let, New Object() {originalFormat})
                                    actualValue = xl(textCell, "Value2", Microsoft.VisualBasic.CallType.Get, New Object() {})
                                    preservedTextCount += 1
                                Finally
                                    releaseCom(textCell)
                                End Try
                            End If
                            If Not valuesMatch(expectedValue, actualValue) Then
                                Throw New System.InvalidOperationException("Exported value differs from its source at " &
                                    columnName(columnOffset + 1) & (rowOffset + 1).ToString() & " for group " & groupName)
                            End If
                        End If
                    Next
                Next
                If preservedTextCount > 0 Then log("TEXT_VALUES_PRESERVED | Group=" & groupName & " | Cells=" & preservedTextCount.ToString())
            End If
            xl(excel, "CutCopyMode", Microsoft.VisualBasic.CallType.Let, New Object() {False})

            currentStep = "Preserve column widths and row heights"
            sourceColumns = xl(sourceSheet, "Columns", Microsoft.VisualBasic.CallType.Get, New Object() {})
            destinationColumns = xl(reportSheet, "Columns", Microsoft.VisualBasic.CallType.Get, New Object() {})
            For index As Integer = 1 To lastColumn
                Dim srcColumn As Object = Nothing
                Dim dstColumn As Object = Nothing
                Try
                    srcColumn = xl(sourceColumns, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {index})
                    dstColumn = xl(destinationColumns, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {index})
                    xl(dstColumn, "ColumnWidth", Microsoft.VisualBasic.CallType.Let,
                        New Object() {xl(srcColumn, "ColumnWidth", Microsoft.VisualBasic.CallType.Get, New Object() {})})
                    xl(dstColumn, "Hidden", Microsoft.VisualBasic.CallType.Let,
                        New Object() {xl(srcColumn, "Hidden", Microsoft.VisualBasic.CallType.Get, New Object() {})})
                Finally
                    releaseCom(dstColumn)
                    releaseCom(srcColumn)
                End Try
            Next
            sourceRows = xl(sourceSheet, "Rows", Microsoft.VisualBasic.CallType.Get, New Object() {})
            destinationRows = xl(reportSheet, "Rows", Microsoft.VisualBasic.CallType.Get, New Object() {})
            For index As Integer = 1 To lastRow
                Dim srcRow As Object = Nothing
                Dim dstRow As Object = Nothing
                Try
                    srcRow = xl(sourceRows, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {index})
                    dstRow = xl(destinationRows, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {index})
                    xl(dstRow, "RowHeight", Microsoft.VisualBasic.CallType.Let,
                        New Object() {xl(srcRow, "RowHeight", Microsoft.VisualBasic.CallType.Get, New Object() {})})
                    xl(dstRow, "Hidden", Microsoft.VisualBasic.CallType.Let,
                        New Object() {xl(srcRow, "Hidden", Microsoft.VisualBasic.CallType.Get, New Object() {})})
                Finally
                    releaseCom(dstRow)
                    releaseCom(srcRow)
                End Try
            Next

            If Not missingSource AndAlso repairEmptyHeaderRow AndAlso sourceHeaders(groupName)(0) = 0 Then
                currentStep = "Restore the empty header row"
                Dim templateSheet As Object = Nothing
                Dim templateRange As Object = Nothing
                Dim destinationHeader As Object = Nothing
                Try
                    templateSheet = xl(sourceSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {headerTemplateName})
                    Dim headerAddress As String = "A" & headerRow.ToString() & ":" & columnName(headerTemplateColumnCount) & headerRow.ToString()
                    templateRange = xl(templateSheet, "Range", Microsoft.VisualBasic.CallType.Get, New Object() {headerAddress})
                    destinationHeader = xl(reportSheet, "Range", Microsoft.VisualBasic.CallType.Get, New Object() {headerAddress})
                    xlMethod(templateRange, "Copy", New Object() {destinationHeader}, New String() {"Destination"})
                    If in_ValuesOnly Then
                        xl(destinationHeader, "Value2", Microsoft.VisualBasic.CallType.Let,
                            New Object() {xl(templateRange, "Value2", Microsoft.VisualBasic.CallType.Get, New Object() {})})
                    End If
                    repairedHeaderCount += 1
                    log("HEADER_ROW_REPAIRED | Group=" & groupName & " | HeaderRow=" & headerRow.ToString() & " | Template=" & headerTemplateName)
                Finally
                    releaseCom(destinationHeader)
                    releaseCom(templateRange)
                    releaseCom(templateSheet)
                End Try
            End If
            If missingSource Then
                Dim warningCell As Object = Nothing
                Try
                    warningCell = xl(reportSheet, "Range", Microsoft.VisualBasic.CallType.Get, New Object() {"A1"})
                    Dim originalTitle As String = System.Convert.ToString(xl(warningCell, "Value2", Microsoft.VisualBasic.CallType.Get, New Object() {}))
                    xl(warningCell, "Value2", Microsoft.VisualBasic.CallType.Let,
                        New Object() {originalTitle & " | SOURCE SHEET MISSING: " & groupName})
                Finally
                    releaseCom(warningCell)
                End Try
            ElseIf sourceDataRows = 0 Then
                emptyGroupCount += 1
                log("EMPTY_SOURCE_DATA | Group=" & groupName & " | Action=HeadersOnly")
            End If

            currentStep = "Replace and name the current report sheet"
            If previousNames.Contains(reportName) Then
                oldSheet = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {reportName})
                deleteSheetSafely(destinationBook, oldSheet, reportSheet)
            End If
            xl(reportSheet, "Name", Microsoft.VisualBasic.CallType.Let, New Object() {reportName})
            currentStep = "Order the report sheets"
            Dim namesNow As System.Collections.Generic.HashSet(Of String) = sheetNames(destinationBook)
            For number As Integer = 4 To 1 Step -1
                Dim name As String = "Report " & number.ToString()
                If Not namesNow.Contains(name) Then Continue For
                Dim firstSheet As Object = Nothing
                Dim movingSheet As Object = Nothing
                Try
                    firstSheet = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {1})
                    If String.Equals(System.Convert.ToString(xl(firstSheet, "Name", Microsoft.VisualBasic.CallType.Get, New Object() {})),
                        name, System.StringComparison.OrdinalIgnoreCase) Then Continue For
                    movingSheet = xl(destinationSheets, "Item", Microsoft.VisualBasic.CallType.Get, New Object() {name})
                    xlMethod(movingSheet, "Move", New Object() {firstSheet}, New String() {"Before"})
                Finally
                    releaseCom(movingSheet)
                    releaseCom(firstSheet)
                End Try
            Next
            Dim resultNames As System.Collections.Generic.HashSet(Of String) = sheetNames(destinationBook)
            If Not resultNames.Contains(reportName) Then Throw New System.InvalidOperationException("The report worksheet is missing before save.")
            Dim expectedCount As Integer = previousNames.Count + If(previousNames.Contains(reportName), 0, 1)
            If resultNames.Count <> expectedCount Then Throw New System.InvalidOperationException("Unexpected final worksheet count: " & fileName)
            For Each previousName As String In previousNames
                If Not resultNames.Contains(previousName) Then Throw New System.InvalidOperationException("An existing worksheet would be lost: " & previousName)
            Next
            log("TRANSFER_VERIFIED | Group=" & groupName & " | Sheet=" & reportName & " | DataRows=" & sourceDataRows.ToString() &
                " | MissingSource=" & missingSource.ToString() & " | FinalSheetCount=" & resultNames.Count.ToString())
            currentStep = "Save the staged group workbook"
            If existed Then
                xl(destinationBook, "Save", Microsoft.VisualBasic.CallType.Method, New Object() {})
            Else
                xl(destinationBook, "SaveAs", Microsoft.VisualBasic.CallType.Method, New Object() {stagedPath, 51})
            End If
            xl(destinationBook, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
            releaseCom(destinationBook)
            destinationBook = Nothing
            If Not System.IO.File.Exists(stagedPath) OrElse (New System.IO.FileInfo(stagedPath)).Length = 0 Then
                Throw New System.IO.IOException("Excel did not create the staged workbook: " & fileName)
            End If
            log("STAGED | Group=" & groupName & " | Sheet=" & reportName)
        Finally
            releaseCom(oldSheet)
            releaseCom(reportSheet)
            releaseCom(anchorSheet)
            releaseCom(destinationRows)
            releaseCom(sourceRows)
            releaseCom(destinationColumns)
            releaseCom(sourceColumns)
            releaseCom(destinationRange)
            releaseCom(sourceRange)
            releaseCom(sourceAreaColumns)
            releaseCom(sourceAreaRows)
            releaseCom(sourceArea)
            releaseCom(sourceSheet)
            releaseCom(sourceSheets)
            releaseCom(destinationSheets)
        End Try
    Next

    ' Close our workbooks and Excel before moving/replacing physical files.
    currentStep = "Close export Excel instance"
    xl(sourceBook, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
    releaseCom(sourceBook)
    sourceBook = Nothing
    xl(groupsBook, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
    releaseCom(groupsBook)
    groupsBook = Nothing
    xl(excel, "Quit", Microsoft.VisualBasic.CallType.Method, New Object() {})
    releaseCom(workbooks)
    workbooks = Nothing
    releaseCom(excel)
    excel = Nothing

    ' Publication is per file, not one transaction across every group.
    ' If a later file fails, rerunning safely replaces this report's sheet again.
    currentStep = "Publish group workbooks"
    For Each groupName As String In groups
        currentGroup = groupName
        Dim fileName As String = "New Report - " & groupName & ".xlsx"
        Dim finalPath As String = System.IO.Path.Combine(periodFolder, fileName)
        Dim stagedPath As String = System.IO.Path.Combine(stageFolder, fileName)
        Dim originalHash As String = expectedHashes(finalPath)
        If originalHash Is Nothing Then
            ' Move does not overwrite a file that unexpectedly appeared.
            System.IO.File.Move(stagedPath, finalPath)
        Else
            If Not System.IO.File.Exists(finalPath) Then
                Throw New System.IO.IOException("Output workbook disappeared during export: " & fileName)
            End If
            Using guard As New System.IO.FileStream(finalPath, System.IO.FileMode.Open,
                        System.IO.FileAccess.ReadWrite, System.IO.FileShare.None)
                If Not String.Equals(hashStream(guard), originalHash, System.StringComparison.Ordinal) Then
                    Throw New System.IO.IOException("Output workbook changed during export. Rerun after resolving the concurrent edit: " & fileName)
                End If
            End Using
            System.IO.Directory.CreateDirectory(backupFolder)
            ' No Delete + overwrite fallback. If Replace is unsupported on the share, fail safely.
            System.IO.File.Replace(stagedPath, finalPath, System.IO.Path.Combine(backupFolder, fileName), False)
        End If
        completedFiles.Add(finalPath)
        log("COMMITTED | Group=" & groupName & " | Sheet=" & reportName)
    Next

    out_UpdatedFiles = completedFiles.ToArray()
    out_Summary = "RunId=" & runId & " | Period=" & reportMonth & " | InputPeriod=" & periodInput & " | " & reportName &
                  " | UpdatedFiles=" & completedFiles.Count.ToString() &
                  " | MissingSheets=" & missingSheetCount.ToString() & " | RepairedHeaders=" & repairedHeaderCount.ToString() &
                  " | EmptyGroups=" & emptyGroupCount.ToString() & " | DuplicateNamesIgnored=" & duplicateCount.ToString()
    log("SUCCESS | UpdatedFiles=" & completedFiles.Count.ToString() & " | MissingSheets=" & missingSheetCount.ToString() &
        " | RepairedHeaders=" & repairedHeaderCount.ToString() & " | EmptyGroups=" & emptyGroupCount.ToString())
    success = True
Catch ex As System.Exception
    Try
        If System.IO.Directory.Exists(journalFolder) Then
            log("FAILED | Group=" & currentGroup & " | Step=" & currentStep & " | Type=" & ex.GetType().FullName &
                " | Message=" & ex.Message.Replace(System.Environment.NewLine, " ") &
                " | CommittedFiles=" & completedFiles.Count.ToString())
        End If
    Catch
        ' Do not hide the original failure if writing the failure log also fails.
    End Try
    Throw
Finally
    ' Cleanup uses only objects created here. Never Kill Process all EXCEL instances.
    For Each book As Object In New Object() {destinationBook, sourceBook, groupsBook}
        If book Is Nothing Then Continue For
        Try
            xl(book, "Close", Microsoft.VisualBasic.CallType.Method, New Object() {False})
        Catch
        Finally
            Try
                releaseCom(book)
            Catch
            End Try
        End Try
    Next
    If excel IsNot Nothing Then
        Try
            xl(excel, "Quit", Microsoft.VisualBasic.CallType.Method, New Object() {})
        Catch
        End Try
    End If
    Try
        releaseCom(workbooks)
    Catch
    End Try
    Try
        releaseCom(excel)
    Catch
    End Try
    If success Then
        Try
            System.IO.Directory.Delete(stageFolder, True)
        Catch
            ' Final files are already published; leftover staging can be removed later.
        End Try
    End If
    If lockStream IsNot Nothing Then lockStream.Dispose()
End Try
